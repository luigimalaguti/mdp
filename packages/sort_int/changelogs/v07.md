#include <cassert>
#include <cstdint>
#include <cstdio>
#include <cstdlib>
#include <utility>

namespace mdp {
    template <typename T>
    int compare(const void *pointer_a, const void *pointer_b) {
        T value_a = *(T *)pointer_a;
        T value_b = *(T *)pointer_b;
        return (value_a > value_b) - (value_a < value_b);
    }

    template <typename T>
    class vector {
    private:
        size_t size_;
        size_t capacity_;
        T *data_;

    public:
        vector(size_t capacity = 0) : size_(0), capacity_(capacity), data_(nullptr) {
            if (capacity_ > 0) {
                data_ = new T[capacity_];
            }
        }

        vector(const vector &other) : size_(other.size_), capacity_(other.capacity_), data_(new T[capacity_]) {
            for (size_t index = 0; index < size_; index++) {
                data_[index] = other.data_[index];
            }
        }

        vector(vector &&other) : size_(other.size_), capacity_(other.capacity_), data_(other.data_) {
            other.data_ = nullptr;
        }

        ~vector() {
            delete[] data_;
        }

        vector &operator=(vector rhs) {
            swap(*this, rhs);
            return *this;
        }

        const T &operator[](size_t index) const {
            return data_[index];
        }

        T &operator[](size_t index) {
            return data_[index];
        }

        void push_back(const T &number) {
            if (size_ >= capacity_) {
                size_t temp_capacity = capacity_ == 0 ? 1 : capacity_ * 2;
                T *temp_data = new T[temp_capacity];
                for (size_t index = 0; index < size_; index++) {
                    temp_data[index] = data_[index];
                }
                delete[] data_;
                data_ = temp_data;
                capacity_ = temp_capacity;
            }
            data_[size_] = number;
            size_++;
        }

        void sort() {
            qsort(data_, size_, sizeof(T), compare<T>);
        }

        size_t size() const {
            return size_;
        }

        const T &at(size_t index) const {
            assert(index < size_);
            return data_[index];
        }

        bool empty() const {
            return size_ == 0;
        }

        friend void swap(vector &vector_a, vector &vector_b) {
            using std::swap;
            swap(vector_a.size_, vector_b.size_);
            swap(vector_a.capacity_, vector_b.capacity_);
            swap(vector_a.data_, vector_b.data_);
        }
    };
}  // namespace mdp

mdp::vector<int32_t> read_from_file(const char *filename, size_t capacity) {
    mdp::vector<int32_t> vec(capacity);
    FILE *file = fopen(filename, "r");
    if (file != NULL) {
        int32_t number;
        while (fscanf(file, "%d", &number) == 1) {
            vec.push_back(number);
        }
        fclose(file);
    }
    return vec;
}

bool write_to_file(const char *filename, const mdp::vector<int32_t> &vec) {
    FILE *file = fopen(filename, "w");
    if (file == NULL) {
        return false;
    }
    for (size_t index = 0; index < vec.size(); index++) {
        fprintf(file, "%d\n", vec.at(index));
    }
    fclose(file);
    return true;
}

int main(int argc, char **argv) {
    if (argc != 3) {
        printf("Usage: %s <input_file> <output_file>\n", argv[0]);
        return EXIT_FAILURE;
    }

    const char *input_filename = argv[1];
    const char *output_filename = argv[2];

    mdp::vector<int32_t> vec;
    vec = read_from_file(input_filename, 10);
    if (vec.empty() == true) {
        printf("Error: Could not open file %s\n", input_filename);
        return EXIT_FAILURE;
    }

    mdp::vector<int32_t> sorted = vec;
    vec.sort();
    swap(vec, sorted);

    bool result = write_to_file(output_filename, sorted);
    if (result == false) {
        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}

## Details

#### Standard copy

In C++ la libreria standard possiede molte funzioni utili e soprattutto efficienti. Una delle prime cose che possiamo fare per concludere la transizione dal linguaggio C al linguaggio C++ è sostituire le funzioni di copia manuale con le funzioni della libreria standard. Per fare ciò possiamo fare uso della funzione `std::copy_n`, la quale permette di svolgere una copia di dati molto a basso livello, con la conseguenza di essere altamente efficiente.

```cpp
namespace mdp {
    // ...

    template <typename T>
    class vector {
        // ...

        vector(const vector &other) : size_(other.size_), capacity_(other.capacity_), data_(new T[capacity_]) {
            std::copy_n(other.data_, size_, data_);
        }

        // ...

        void push_back(const T &number) {
            if (size_ >= capacity_) {
                size_t temp_capacity = capacity_ == 0 ? 1 : capacity_ * 2;
                T *temp_data = new T[temp_capacity];
                std::copy_n(data_, size_, temp_data);
                delete[] data_;
                data_ = temp_data;
                capacity_ = temp_capacity;
            }
            data_[size_] = number;
            size_++;
        }

        // ...
    };
}  // namespace mdp

// ...
```

La sua firma è molto semplice. Richiede tre parametri, il puntatore alla sorgente, il numero di elementi da copiare e il puntatore alla destinazione. In questo modo possiamo sostituire le copie manuali con una copia standard, più efficiente e più leggibile.

#### Standard input stream

Il linguaggio C++ mette a disposizione una libreria standard `fstream` che consente di gestire i file in maniera più semplice e intuitiva rispetto al linguaggio C. In particolare, possiamo utilizzare le classi `std::ifstream` e `std::ofstream` per leggere e scrivere file rispettivamente. Queste classi forniscono un'interfaccia più sicura e moderna per la gestione dei file.

```cpp
// ...

mdp::vector<int32_t> read_from_file(const char *filename, size_t capacity) {
    mdp::vector<int32_t> vec(capacity);
    std::ifstream is(filename);
    if (is) {
        int32_t number;
        while (is >> number) {
            vec.push_back(number);
        }
    }
    return vec;
}

// ...
```

A differenza del C dove utilizzavamo `fopen` sia per lettura che scrittura su file e poi specificarne il tipo di accesso, in C++ abbiamo a disposizione una gestione dei file orientata alle classi. Quindi `std::ifstream` è una classe che rappresenta un flusso di input da un file. L'inizializzazione di un oggetto `std::ifstream` con il nome del file apre automaticamente il file in modalità lettura. Accetta opzionalmente un secondo parametro che specifica la modalità di apertura del file, se volessimo aprirlo in modalità binaria dovremo specificare `std::ios::binary` come secondo parametro. In questo caso, non è necessario specificare la modalità di apertura del file poiché il file viene aperto in modalità testo per default.

Noi abbiamo definito e inizializzato tutto in una volta sola la variabile `is`. Questo prende il nome di **RAII**, **Resource Acquisition Is Initialization**, una tecnica di programmazione che garantisce che le risorse vengano acquisite e rilasciate in modo sicuro. In questo caso, l'oggetto `is` acquisisce la risorsa del file e la rilascia automaticamente quando esce dallo scope, evitando perdite di memoria o file aperti. Infatti non è necessario chiudere il file manualmente con `is.close()`, poiché il distruttore della classe `std::ifstream` si occupa di chiudere automaticamente il file quando l'oggetto `is` esce dallo scope.

L'alternativa classica sarebbe stata quella di dichiarare la variabile `is` senza inizializzarla e poi aprire il file con il metodo `open()`. In questo caso, avremmo dovuto chiudere manualmente il file con il metodo `close()`.

La lettura da file, invece, non richiede più l'uso di `fscanf`, ma possiamo utilizzare l'operatore di estrazione `>>` per leggere i dati dal file. Questo operatore gestisce automaticamente la conversione dei dati e la gestione degli spazi bianchi, rendendo il codice più leggibile e meno soggetto a errori. In questo caso l'operatore `>>` svolge lo stesso esatto compito di `fscanf`, ma in maniera più sicura e moderna.

Una alternativa sarebbe stato il metodo `get()` di `is`. Questo metodo legge un singolo byte dal file e lo memorizza in una variabile. Tuttavia, per leggere interi, l'operatore di estrazione `>>` è più appropriato e conveniente.

#### Standard output stream

In modo analogo a quanto visto per lo stream in input, possiamo fare la stessa cosa anche per lo stream in output.

```cpp
// ...

bool write_to_file(const char *filename, const mdp::vector<int32_t> &vec) {
    std::ofstream os(filename);
    if (!os) {
        return false;
    }
    for (size_t index = 0; index < vec.size(); index++) {
        os << vec[index] << "\n";
    }
    return true;
}

// ...
```

Il processo è simile a quello della lettura da file. In questo caso, utilizziamo la classe `std::ofstream` per aprire il file in modalità scrittura. L'oggetto `os` acquisisce la risorsa del file e la rilascia automaticamente quando esce dallo scope, evitando perdite di memoria o file aperti. Anche in questo caso, non è necessario chiudere il file manualmente con `os.close()`, poiché il distruttore della classe `std::ofstream` si occupa di chiudere automaticamente il file quando l'oggetto `os` esce dallo scope.

La scrittura su file avviene anche in questo caso tramite l'operatore di inserimento `<<`, che gestisce automaticamente la conversione dei dati e la gestione degli spazi bianchi, rendendo il codice più leggibile e meno soggetto a errori. In questo caso l'operatore `<<` svolge lo stesso esatto compito di `fprintf`, ma in maniera più sicura e moderna.

Una alternativa rispetto l'operatore `<<` sarebbe stato il metodo `std::println()`, come si può notare nel codice del `main`. Questo metodo permette di stampare una stringa formattata su un flusso di output, che sia standard output o un file.

```cpp
// ...

int main(int argc, char **argv) {
    if (argc != 3) {
        std::println("Usage: {} <input_file> <output_file>", argv[0]);
        return EXIT_FAILURE;
    }

    const char *input_filename = argv[1];
    const char *output_filename = argv[2];

    mdp::vector<int32_t> vec;
    vec = read_from_file(input_filename, 10);
    if (vec.empty() == true) {
        std::println("Error: Could not open file {}", input_filename);
        return EXIT_FAILURE;
    }

    mdp::vector<int32_t> sorted = vec;
    vec.sort();
    swap(vec, sorted);

    bool result = write_to_file(output_filename, sorted);
    if (result == false) {
        std::println("Error: Could not open file {}", output_filename);
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

La differenza principale tra l'operatore `<<` e il metodo `std::println()` è che quest'ultimo permette di stampare una stringa formattata, mentre l'operatore `<<` stampa i dati così come sono. In questo caso, il metodo `std::println()` è più conveniente per stampare messaggi di errore o informazioni di debug, poiché consente di includere variabili all'interno della stringa senza doverle convertire manualmente in stringhe.

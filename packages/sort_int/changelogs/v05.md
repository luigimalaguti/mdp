```cpp
#include <cassert>
#include <cstdint>
#include <cstdio>
#include <cstdlib>

int compare_int32(const void *pointer_a, const void *pointer_b) {
    int32_t number_a = *(int32_t *)pointer_a;
    int32_t number_b = *(int32_t *)pointer_b;
    return (number_a > number_b) - (number_a < number_b);
}

struct vector {
    size_t size_;
    size_t capacity_;
    int32_t *data_;

    vector() {
        size_ = 0;
        capacity_ = 0;
        data_ = nullptr;
    }

    vector(size_t capacity) {
        size_ = 0;
        capacity_ = capacity;
        data_ = new int32_t[capacity];
    }

    vector(const vector &other) {
        size_ = other.size_;
        capacity_ = other.capacity_;
        data_ = new int32_t[capacity_];
        for (size_t index = 0; index < size_; index++) {
            data_[index] = other.data_[index];
        }
    }

    vector(vector &&other) {
        size_ = other.size_;
        capacity_ = other.capacity_;
        data_ = other.data_;
        other.data_ = nullptr;
    }

    ~vector() {
        delete[] data_;
    }

    vector &operator=(const vector &rhs) {
        if (this != &rhs) {
            size_ = rhs.size_;
            capacity_ = rhs.capacity_;
            delete[] data_;
            data_ = new int32_t[capacity_];
            for (size_t index = 0; index < size_; index++) {
                data_[index] = rhs.data_[index];
            }
        }
        return *this;
    }

    vector &operator=(vector &&rhs) {
        size_ = rhs.size_;
        capacity_ = rhs.capacity_;
        delete[] data_;
        data_ = rhs.data_;
        rhs.data_ = nullptr;
        return *this;
    }

    void push_back(int32_t number) {
        if (size_ >= capacity_) {
            size_t temp_capacity = capacity_ == 0 ? 1 : capacity_ * 2;
            int32_t *temp_data = new int32_t[temp_capacity];
            for (size_t index = 0; index < size_; index++) {
                temp_data[index] = data_[index];
            }
            delete[] data_;
            data_ = temp_data;
            capacity_ = temp_capacity;
        }
        data_[size_] = number;
        size_++;
    }

    void sort() {
        qsort(data_, size_, sizeof(int32_t), compare_int32);
    }

    size_t size() const {
        return size_;
    }

    int32_t at(size_t index) const {
        assert(index < size_);
        return data_[index];
    }

    bool empty() const {
        return size_ == 0;
    }
};

vector read_from_file(const char *filename, size_t capacity) {
    vector vec(capacity);
    FILE *file = fopen(filename, "r");
    if (file != NULL) {
        int32_t number;
        while (fscanf(file, "%d", &number) == 1) {
            vec.push_back(number);
        }
        fclose(file);
    }
    return vec;
}

bool write_to_file(const char *filename, const vector &vec) {
    FILE *file = fopen(filename, "w");
    if (file == NULL) {
        return false;
    }
    for (size_t index = 0; index < vec.size(); index++) {
        fprintf(file, "%d\n", vec.at(index));
    }
    fclose(file);
    return true;
}

int main(int argc, char **argv) {
    if (argc != 3) {
        printf("Usage: %s <input_file> <output_file>\n", argv[0]);
        return EXIT_FAILURE;
    }

    const char *input_filename = argv[1];
    const char *output_filename = argv[2];

    vector vec;
    vec = read_from_file(input_filename, 10);
    if (vec.empty() == true) {
        printf("Error: Could not open file %s\n", input_filename);
        return EXIT_FAILURE;
    }

    vec.sort();

    bool result = write_to_file(output_filename, vec);
    if (result == false) {
        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

## Details

### Copy and swap idiom

L'ultima versione del codice ci ha permesso evitare diverse copie in eccesso della struttura `vector` grazie all'implementazione del move constructor e del move assignment operator. Infatti questi due nuovi metodi evitano copie degli oggetti temporanei, ovvero oggetti che verrebbero distrutti subito dopo la loro creazione. L'idea di fondo era quindi di entrare in possesso di questo contenuto, anziché copiarlo, in quanto non più necessario.

Tuttavia, guardando le implementazioni dei costruttori, copy e move, insieme agli operatori, sempre copy e move, possiamo notare una certa duplicazione di codice. In particolare, il copy assignment operator ripete quasi identica la logica del copy constructor, e il move assignment operator ripete quasi identica la logica del move constructor. L'unica differenza è che l'assignment deve anche occuparsi di liberare la memoria precedentemente posseduta dall'oggetto. Questa duplicazione potrebbe generare problemi in futuro, se la logica di costruzione di `vector` cambiasse, dovremmo ricordarci di aggiornarla in quattro punti diversi, con il rischio concreto di disallineare accidentalmente uno di essi.

Per risolvere questo problema, possiamo usare il cosiddetto **copy-and-swap idiom**, che ci permette di eliminare questa duplicazione, riusando i costruttori che abbiamo già scritto anche per implementare l'assegnamento.

Supponiamo quindi di voler scrivere una funzione di supporto che si occupi di scambiare i contenuti di due oggetti `vector`. Questa funzione richiederà ovviamente due parametri di tipo `vector &` e non dovrà restituire nessun valore in particolare.

```cpp
// ...

struct vector {
    // ...
};

// ...

void swap(vector &vector_a, vector &vector_b) {
    size_t temp_size = vector_a.size_;
    vector_a.size_ = vector_b.size_;
    vector_b.size_ = temp_size;
    size_t temp_capacity = vector_a.capacity_;
    vector_a.capacity_ = vector_b.capacity_;
    vector_b.capacity_ = temp_capacity;
    int32_t *temp_data = vector_a.data_;
    vector_a.data_ = vector_b.data_;
    vector_b.data_ = temp_data;
}

// ...
```

A questo punto potremmo provare ad applicarla a un qualche metodo del nostro `vector` per cercare di eliminare la duplicazione di codice. Proviamo quindi a partire con il metodo `vector &operator=(const vector &rhs)`. Questo metodo è il copy assignment operator, il quale ha l'obbiettivo di ricevere un riferimento a un `vector` per poi copiarne il suo contenuto all'interno del left hand side `vector`, ovvero `this`. Quindi potremmo modificare il suo codice come segue.

```cpp
// ...

struct vector {
    // ...

    vector &operator=(const vector &rhs) {
        vector temp_vec = rhs;
        swap(*this, temp_vec);
        return *this;
    }

    // ...
};

// ...
```

L'assignment operator riceve il riferimento del right hand side `vector`, creiamo una copia in un vettore temporaneo `temp_vec` attraverso il copy constructor e poi utilizziamo la funzione di supporto `swap` per scambiare i contenuti di `this` e `temp_vec`. In questo modo, il contenuto di `temp_vec`, che è una copia del right hand side, viene messo in `this`, mentre il contenuto precedente di `this` viene spostato in `temp_vec`, che verrà distrutto alla fine della funzione, liberando così la memoria precedentemente occupata da `this`. Essenzialmente lo stesso comportamento del copy assignment operator originale, quindi deallocazione della memoria e copia dei dati, ma in questo caso la copia avviene attraverso il copy constructor e non manualmente dal nostro codice.

Ora però possiamo fare un osservazione. Se all'interno del body del copy assignment operator stiamo già creando un oggetto temporaneo `temp_vec` che è una copia del right hand side, allora potremmo accettare direttamente come parametro della funzione un oggetto `vector` passato per valore, invece che per riferimento.

```cpp
// ...

struct vector {
    // ...

    vector &operator=(vector rhs) {
        swap(*this, rhs);
        return *this;
    }

    // ...
};

// ...
```

In questo modo effettueremo lo scambio del contenuto sempre da un oggetto `vector` che risulti già una copia del right hand side `vector`. Importante notare nuovamente che non stiamo applicando una copia aggiuntiva rispetto la precedente versione del codice, ma stiamo semplicemente spostando la copia del right hand side `vector` all'interno del parametro `rhs`, che è un oggetto locale della funzione. In questo modo, il copy assignment operator svolge la copia che avremmo fatto manualmente prima, ma in maniera più elegante e senza duplicazione di codice.

Ma ora arriva la parte interessante del copy-and-swap idiom. Non è più necessario definire il move assignment operator. Esatto, proprio così. L'assignment operator per come lo abbiamo appena definito, accetta un oggetto `vector` passato per valore. Questo significa che se passiamo un oggetto temporaneo, come ad esempio il risultato di una funzione che restituisce un `vector`, allora il compilatore utilizzerà il move constructor per costruire l'oggetto `rhs` all'interno del copy assignment operator. In questo modo, il move assignment operator non è più necessario, in quanto il copy assignment operator si occupa già di gestire sia i casi di copia che di spostamento.

Per avere la prova del suo funzionamento e della sua efficacia possiamo provare a utilizzare il debugger. Ma appena proviamo a compilare il codice, ci accorgiamo che il compilatore ci restituisce un errore. L'errore è dovuto al fatto che `operator=`, definito dentro `vector`, chiama `swap`. Ma nel punto in cui il compilatore legge quel corpo, `swap` non è ancora stata dichiarata da nessuna parte nel file. La sua definizione compare più avanti, dopo la chiusura di `struct vector`. Il compilatore processa il file dall'alto verso il basso, e un nome dev'essere visibile, cioè dichiarato, prima di poter essere usato. Questo è un classico problema di ordine di dichiarazione, e si risolve con una forward declaration.

Per risolvere questo problema, dobbiamo forward-dichiarare sia la struttura `vector` sia la funzione `swap` prima della definizione completa di `vector`. Una dichiarazione anticipata di `struct vector` permette al prototipo di `swap` di usare `vector &` come tipo di parametro, un tipo incompleto è sufficiente per un riferimento, mentre il prototipo di `swap` permette al corpo di `operator=`, definito dentro `vector`, di trovare `swap` per nome. Le definizioni complete restano dove sono già, prima l'intera struttura `vector`, poi il corpo di `swap`.

```cpp
struct vector;
void swap(vector &vector_a, vector &vector_b);

// ...

struct vector {
    // ...

    vector &operator=(vector rhs) {
        swap(*this, rhs);
        return *this;
    }

    // ...
};

void swap(vector &vector_a, vector &vector_b) {
    // ...
}

// ...
```

Guardando questo trace, notiamo che `vec03`, il vettore creato tramite move constructor al momento del `return vec` di `read_from_file`, diventa direttamente il parametro `rhs` di `operator=`, senza nessuna costruzione intermedia. A prima vista potrebbe sembrare che il compilatore abbia semplicemente evitato una copia superflua nel passaggio del parametro. In realtà, quello che stiamo osservando è di nuovo la *Guaranteed Copy Elision*, la stessa ottimizzazione analizzata nella precedente versione.

Questo potrebbe sembrare in contraddizione con quanto stabilito allora, dove avevamo verificato che un'assegnazione non beneficia mai della Guaranteed Copy Elision, a differenza di un'inizializzazione. E infatti non lo è, l'espressione `vec = read_from_file(...)` nel suo complesso resta un'assegnazione, e come tale non ne beneficia. Ma il punto della Guaranteed Copy Elision non è più tra il valore restituito da `read_from_file` e la variabile `vec` del `main`, come accadeva nella precedente versione. Ora `operator=` accetta il proprio parametro `rhs` per valore, e costruire quel parametro a partire dal valore restituito da `read_from_file` è, a tutti gli effetti, un'inizializzazione. L'inizializzazione del parametro `rhs` a partire da un oggetto temporaneo senza nome. È questa inizializzazione, non l'assegnazione esterna, a beneficiare della garanzia introdotta dal C++17.

Per verificarlo, possiamo disabilitare anche questa ottimizzazione. Compilando con `-std=c++11 -fno-elide-constructors` invece che con `-std=c++23 -fno-elide-constructors`, torniamo a uno standard in cui la Guaranteed Copy Elision non è ancora obbligatoria, e quindi anche il passaggio del parametro può essere disattivato.

Ora la differenza è visibile, compare un secondo `vector(vector &&other)`, che costruisce `vec04` a partire da `vec03`. Questa è esattamente la costruzione che la Guaranteed Copy Elision, nel trace precedente, ci aveva permesso di evitare. Senza di essa, il compilatore deve prima materializzare il valore di ritorno di `read_from_file` come oggetto a sé stante, `vec03`, e solo in un secondo momento, con una costruzione separata, usarlo per inizializzare il parametro `rhs`, `vec04`. Il costruttore chiamato per questo secondo passaggio è comunque il move constructor, e non il copy constructor, perché `vec03` è un oggetto temporaneo, dimostrando così che il nostro `operator=(vector rhs)` gestisce correttamente sia la copia sia lo spostamento, a seconda del tipo di argomento che riceve, con un'unica implementazione.

In questo modo, il copy-and-swap idiom ci permette di gestire sia i casi di copia che di spostamento senza duplicazione di codice, un solo `operator=`, i cui comportamenti da "copia" o "sposta" emergono automaticamente dalla scelta del costruttore usato per il suo parametro, esattamente come già accadeva per la risoluzione degli overload vista in v04.

### Classes

Avendo compreso il copy-and-swap idiom, e avendo constatato che il move assignment operator non è più necessario, possiamo ora concentrarci su un altro aspetto del nostro codice, abbandonare l'utilizzo di `struct` per la definizione di `vector` e passare a `class`. Infatti, sebbene fino a questo momento abbiamo utilizzato `struct` per definire degli oggetti, in C++ esiste anche la possibilità di utilizzare `class`. Per introdurla proviamo immediatamente a modificare il nostro codice.

```cpp
class vector;
void swap(vector &vector_a, vector &vector_b);

// ...

class vector {
    // ...
};

// ...
```

Un mare di errori da parte del compilatore ci avvisa che non possiamo più accedere ai membri di `vector` come prima. A giudicare dal numero di errori verrebbe da pensare che sia necessario svolgere un grosso lavoro di refactoring, ma in realtà non è così. Il motivo è molto banale e facilmente risolvibile.

La differenza tra definire un oggetto con `struct` o con `class` risiede semplicemente nella visibilità dei membri. In una `struct`, i membri sono pubblici per default, come sappiamo anche dal linguaggio C. In una `class`, invece, tutti i membri sono privati per default. Questo significa che, nel nostro caso, tutti i membri di `vector` sono diventati privati e quindi non accessibili dall'esterno della classe. Per risolvere questo problema ci basta utilizzare i modificatori di accesso `public`, `protected` e `private`. In particolare, per rendere i membri di `vector` accessibili dall'esterno della classe, dobbiamo dichiararli come `public`.

```cpp
// ...

class vector {
private:
    size_t size_;
    size_t capacity_;
    int32_t *data_;

public:
    vector() {
        size_ = 0;
        capacity_ = 0;
        data_ = nullptr;
    }

    // ...
};

// ...
```

In questo caso, la keyword `private` non è strettamente necessaria, in quanto i membri sono privati per default, ma la sua presenza rende più chiaro il codice. Invece, la keyword `public` è necessaria per rendere i membri accessibili dall'esterno della classe. Una nota interessante riguarda al fatto che utilizziamo i modificatori di accesso sempre in modo esplicito, utilizzare `class` o `struct` non presenta più alcuna differenza. Infatti, come la `struct`, anche per `class` la sua dimensione reale in memoria risulta essere composta solamente dagli attributi della classe, i metodi definiti all'interno della classe non occupano spazio in memoria, proprio come per la `struct`.

#### Friend swap function

Con questa correzione dei modificatori di accesso per `class vector` abbiamo risolto la maggior parte dei problemi riscontrati, tutti eccetto quelli mostrati nella funzione `swap`.

Il motivo dell'errore anche in questo caso è semplice. Avendo definito gli attributi di `vector` come privati, la funzione `swap` non riesce più ad accedervi, non essendo un metodo della classe. La prima soluzione che potrebbe venire in mente è quella di definire tutti i getter e setter di cui abbiamo bisogno. Purtroppo questa non è una buona idea, in quanto andremmo a violare il principio di incapsulamento, rendendo pubblici attributi che dovrebbero rimanere privati.

Il linguaggio C++ permette di utilizzare un meccanismo tale per cui si indica alla classe che esiste una funzione esterna che può accedere ai suoi membri privati. Questo meccanismo si chiama `friend`. In questo caso, possiamo dichiarare la funzione `swap` come `friend` della classe `vector`, in modo che possa accedere ai suoi membri privati.

```cpp
// ...

class vector {
private:
    size_t size_;
    size_t capacity_;
    int32_t *data_;

public:
    // ...

    friend void swap(vector &vector_a, vector &vector_b);
};

void swap(vector &vector_a, vector &vector_b) {
    // ...
}

// ...
```

Attraverso questo meccanismo, la funzione `swap` può accedere correttamente agli attributi privati della classe `vector`, seppur non essendo un metodo della classe stessa. Infatti, la definizione di `friend void swap(vector &vector_a, vector &vector_b)` all'interno della classe non vuole indicare che la funzione `swap` sia un metodo della classe, ma semplicemente una funzione che ha il permesso di accedere ai suoi membri privati. Questo rimane corretto anche se definissimo direttamente il body di `swap` direttamente all'interno della classe, anche se non strettamente necessario. Anche in questo caso, `swap` non diventa un metodo della classe, ma rimane una funzione esterna che ha il permesso di accedere ai membri privati della classe. Inoltre, definendo la funzione `swap` come `friend` di `vector`, possiamo eliminare le forward declaration di `struct vector` e `void swap(vector &vector_a, vector &vector_b)`, in quanto la funzione `swap` è già visibile all'interno della classe `vector`.

```cpp
// ...

class vector {
private:
    size_t size_;
    size_t capacity_;
    int32_t *data_;

public:
    // ...

    friend void swap(vector &vector_a, vector &vector_b) {
        // ...
    }
};

// ...
```

#### Namespaces

Fino ad ora `vector`, `swap` e le altre funzioni di supporto vivono tutte nel namespace globale. Questo va bene per un piccolo programma dimostrativo come il nostro, ma diventa un problema in un progetto più grande. `vector` e `swap` sono nomi estremamente comuni e il rischio di collisione con altro codice o altre librerie è concreto. Il C++ risolve questo problema con i namespace, che ci permettono di raggruppare le nostre definizioni sotto un nome distintivo, evitando conflitti.

```cpp
namespace mdp {
    int compare_int32(const void *pointer_a, const void *pointer_b) {
        int32_t number_a = *(int32_t *)pointer_a;
        int32_t number_b = *(int32_t *)pointer_b;
        return (number_a > number_b) - (number_a < number_b);
    }

    class vector {
        // ...
    };
}  // namespace mdp

// ...
```

Ma applicando questo `namespace` appena definito, notiamo come il compilatore ci restituisca un errore. Infatti, gli errori vengono principalmente evidenziati in prossimità del tipo di dato `vector`. Il motivo riguarda al fatto che il compilatore con questo cambiamento non è più in grado di trovare il tipo `vector` all'interno del namespace globale, in quanto ora si trova all'interno del namespace `mdp`.

Per risolvere questo problema, dobbiamo specificare il namespace corretto quando facciamo riferimento a `vector`, ad esempio `mdp::vector`.

```cpp
// ...

mdp::vector read_from_file(const char *filename, size_t capacity) {
    mdp::vector vec(capacity);

    // ...
}

bool write_to_file(const char *filename, const mdp::vector &vec) {
    // ...
}

int main(int argc, char **argv) {
    // ...

    mdp::vector vec;
    vec = read_from_file(input_filename, 10);
    if (vec.empty() == true) {
        printf("Error: Could not open file %s\n", input_filename);
        return EXIT_FAILURE;
    }

    // ...
}
```

In questo modo, il compilatore trova correttamente la classe `vector` all'interno del namespace `mdp`, e il codice compila senza errori.

#### Argument Dependent Lookup

Proviamo a questo punto a fare una piccola modifica nel `main`. Vogliamo creare un secondo vettore in modo da avere la versione ordinata e non.

```cpp
// ...

int main(int argc, char **argv) {
    // ...

    mdp::vector sorted = vec;
    vec.sort();
    swap(vec, sorted);

    bool result = write_to_file(output_filename, sorted);
    if (result == false) {
        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

In questo nuovo codice notiamo un comportamento strano, la funzione `swap` non richiede il namespace `mdp::`, come invece accade per `vector`. Il processo che il compilatore utilizza per risolvere il nome di una funzione o di un tipo è chiamato *name lookup*, ed esistono due varianti principali.

La **lookup qualificata** è quella che usiamo scrivendo `mdp::vector`, diciamo esplicitamente al compilatore in quale namespace cercare il nome, e lui guarda solamente lì. La **lookup non qualificata**, quella che avviene scrivendo semplicemente `swap(vec, sorted)` senza alcun prefisso, è più articolata. Il compilatore cerca prima negli scope in cui ci troviamo, dentro `main`, poi risalendo verso il namespace globale, ma esegue anche un secondo passaggio chiamato **Argument Dependent Lookup**, o **ADL**. L'ADL guarda il tipo degli argomenti passati alla funzione, in questo caso `mdp::vector`, e cerca automaticamente anche nel namespace in cui quel tipo è stato dichiarato, cioè `mdp`, indipendentemente da dove si trovi il codice chiamante. Non serve nessun `using namespace mdp`, è sufficiente che gli argomenti siano di quel tipo.

Per verificarlo, proviamo a chiamare `swap` in modo esplicitamente qualificato.

```cpp
// ...

int main(int argc, char **argv) {
    // ...

    mdp::vector sorted = vec;
    vec.sort();
    mdp::swap(vec, sorted);

    // ...
}
```

Sorprendentemente, questa riga **non compila**, producendo l'errore `error: 'swap' is not a member of 'mdp'`.

Questo può sembrare contraddittorio, `swap` è definita dentro `mdp`, eppure il compilatore sostiene che non ne è un membro. La spiegazione risiede in *come* abbiamo definito `swap`. Ricordiamo la definizione della funzione `swap`, interamente scritta come `friend` dentro la classe `vector`:

```cpp
namespace mdp {
    // ...

    class vector {
        // ...

        friend void swap(vector &vector_a, vector &vector_b) {
            // ...
        }
    };
}  // namespace mdp

// ...
```

Quando una funzione viene dichiarata `friend` per la prima volta dentro una classe, senza che esista già un'altra dichiarazione della stessa funzione a livello di namespace, il C++ stabilisce che quel nome venga sì associato al namespace che racchiude la classe, `mdp` nel nostro caso, ma **esclusivamente ai fini dell'ADL**. Il nome non viene inserito nella tabella di lookup ordinaria di quel namespace. È come se `swap` fosse un post-it attaccato alla classe vector, anziché una voce scritta nell'indice ufficiale di `mdp`, lo trovi solo passando un argomento di tipo `vector`, l'ADL segue proprio il tipo degli argomenti, ma non lo trovi cercandolo per nome dentro il namespace, perché lì non è mai stato effettivamente scritto. Questo meccanismo è noto come **friend name injection**, e la tecnica nel suo complesso viene spesso chiamata *hidden friend idiom*.

Se invece avessimo definito `swap` come una funzione autonoma a livello di namespace, non interamente dentro la classe, ad esempio così:

```cpp
namespace mdp {
    // ...

    class vector {
        // ...

        friend void swap(vector &vector_a, vector &vector_b);
    };

    void swap(vector &vector_a, vector &vector_b) {
        // ...
    }
}  // namespace mdp

// ...
```

Allora `mdp::swap(vec, sorted)` avrebbe compilato senza problemi, perché quella riga sarebbe un membro a pieno titolo del namespace `mdp`, trovabile sia con la lookup ordinaria sia con l'ADL.

Questa non è una limitazione, ma una scelta di design deliberata, ed è diffusissima nel C++ moderno, incluso nella standard library. Il vantaggio principale riguarda quello che potremmo chiamare "inquinamento del namespace". Se `swap` fosse una funzione ordinaria di `mdp`, chiunque scrivesse `using namespace mdp` nel proprio codice si ritroverebbe `mdp::swap(vector &, vector &)` visibile per **qualsiasi** chiamata a `swap(...)`, anche su tipi completamente estranei a `vector`, aumentando inutilmente il numero di candidati che il compilatore deve considerare, e producendo messaggi di errore confusi quando la chiamata coinvolge tipi non correlati. Con lo hidden friend, invece, `swap` di `vector` entra in gioco solo quando gli argomenti sono davvero di tipo `vector`, resta "agganciato" al tipo a cui appartiene, invece di disperdersi nel namespace come una funzione generica disponibile per chiunque.

#### Standard swap

Dal momento che abbiamo definito il nostro namespace `mdp`, possiamo sfruttare la funzione generica `std::swap`, definita dalla libreria standard, per scambiare i singoli membri di `vector`, invece di farlo manualmente. Questa funzione è generica, e può essere utilizzata per scambiare due oggetti di qualsiasi tipo, purché quel tipo supporti la copia o lo spostamento.

```cpp
#include <utility>

namespace mdp {
    // ...

    class vector {
        // ...

        friend void swap(vector &vector_a, vector &vector_b) {
            std::swap(vector_a.size_, vector_b.size_);
            std::swap(vector_a.capacity_, vector_b.capacity_);
            std::swap(vector_a.data_, vector_b.data_);
        }
    }
}  // namespace mdp

// ...
```

Qualificare esplicitamente `std::swap` funziona, ma c'è un modo più idiomatico. Proviamo a scrivere semplicemente `swap(vector_a.size_, vector_b.size_)`, senza `std::` davanti.

Questa riga **non compila**, con l'errore `'swap' was not declared in this scope`. La ragione è che `size_` e `capacity_` sono di tipo `size_t`, e `data_` è un puntatore, tutti tipi fondamentali del C++, che **non hanno alcun namespace associato**. L'ADL, come abbiamo visto, si attiva seguendo il namespace del tipo degli argomenti, ma un tipo fondamentale non appartiene a nessun namespace, quindi l'ADL non ha nulla da seguire fino a `std`, e la nostra `mdp::vector::swap` non viene nemmeno considerata, dato che nessuno degli argomenti è di tipo `vector`. Il compilatore, di fatto, non trova alcuna funzione `swap` candidata.

Per risolvere questo, possiamo portare `std::swap` nella lookup ordinaria locale con una *using-declaration*:

```cpp
namespace mdp {
    // ...

    class vector {
        // ...

        friend void swap(vector &vector_a, vector &vector_b) {
            using std::swap;
            swap(vector_a.size_, vector_b.size_);
            swap(vector_a.capacity_, vector_b.capacity_);
            swap(vector_a.data_, vector_b.data_);
        }
    }
}  // namespace mdp

// ...
```

`using std::swap` non "abilita l'ADL", come si potrebbe pensare. Al contrario, compensa il fatto che l'ADL non può fare nulla per tipi come `size_t` o un puntatore, inserendo `std::swap` come alternativa sempre disponibile nella lookup ordinaria. Il vantaggio di questo pattern, rispetto a scrivere direttamente `std::swap(...)` ovunque, è che se in futuro uno dei membri di `vector` diventasse un tipo classe con il proprio `swap` personalizzato, magari definito come abbiamo appena visto per `vector` stesso come hidden friend, l'ADL lo troverebbe e lo preferirebbe automaticamente rispetto a `std::swap`, senza dover cambiare nulla in questa funzione. Questo pattern, `using std::swap` e `swap(a, b)`, è talmente diffuso nel C++ moderno da avere un nome proprio, **swap idiom**, o *two-step lookup*, ed è esattamente la tecnica che stiamo usando nel codice finale.

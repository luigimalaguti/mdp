```cpp
#include <cstdio>

int compare_int32(const void *pointer_a, const void *pointer_b) {
    int32_t number_a = *(int32_t *)pointer_a;
    int32_t number_b = *(int32_t *)pointer_b;
    return (number_a > number_b) - (number_a < number_b);
}

struct vector {
    size_t size_;
    size_t capacity_;
    int32_t *data_;

    vector() {
        size_ = 0;
        capacity_ = 0;
        data_ = NULL;
    }

    ~vector() {
        free(data_);
    }

    void push_back(int32_t number) {
        if (size_ >= capacity_) {
            size_t temp_capacity = capacity_ == 0 ? 1 : capacity_ * 2;
            int32_t *temp_data = (int32_t *)realloc(data_, temp_capacity * sizeof(int32_t));
            data_ = temp_data;
            capacity_ = temp_capacity;
        }
        data_[size_] = number;
        size_++;
    }

    void sort() {
        qsort(data_, size_, sizeof(int32_t), compare_int32);
    }

    size_t size() const {
        return size_;
    }

    int32_t at(size_t index) const {
        assert(index < size_);
        return data_[index];
    }
};
```

```cpp
#include <cassert>
#include <cstdint>
#include <cstdio>
#include <cstdlib>

int compare_int32(const void *pointer_a, const void *pointer_b) {
    int32_t number_a = *(int32_t *)pointer_a;
    int32_t number_b = *(int32_t *)pointer_b;
    return (number_a > number_b) - (number_a < number_b);
}

struct vector {
    size_t size_;
    size_t capacity_;
    int32_t *data_;

    vector() {
        size_ = 0;
        capacity_ = 0;
        data_ = NULL;
    }

    vector(size_t capacity) {
        size_ = 0;
        capacity_ = capacity;
        data_ = new int32_t[capacity];
    }

    vector(const vector &other) {
        size_ = other.size_;
        capacity_ = other.capacity_;
        data_ = new int32_t[capacity_];
        for (size_t index = 0; index < size_; index++) {
            data_[index] = other.data_[index];
        }
    }

    ~vector() {
        delete[] data_;
    }

    vector &operator=(const vector &rhs) {
        if (this != &rhs) {
            size_ = rhs.size_;
            capacity_ = rhs.capacity_;
            delete[] data_;
            data_ = new int32_t[capacity_];
            for (size_t index = 0; index < size_; index++) {
                data_[index] = rhs.data_[index];
            }
        }
        return *this;
    }

    void push_back(int32_t number) {
        if (size_ >= capacity_) {
            size_t temp_capacity = capacity_ == 0 ? 1 : capacity_ * 2;
            int32_t *temp_data = new int32_t[temp_capacity];
            for (size_t index = 0; index < size_; index++) {
                temp_data[index] = data_[index];
            }
            delete[] data_;
            data_ = temp_data;
            capacity_ = temp_capacity;
        }
        data_[size_] = number;
        size_++;
    }

    void sort() {
        qsort(data_, size_, sizeof(int32_t), compare_int32);
    }

    size_t size() const {
        return size_;
    }

    int32_t at(size_t index) const {
        assert(index < size_);
        return data_[index];
    }

    bool empty() const {
        return size_ == 0;
    }
};

vector read_from_file(const char *filename, size_t capacity) {
    vector vec(capacity);
    FILE *file = fopen(filename, "r");
    if (file != NULL) {
        int32_t number;
        while (fscanf(file, "%d", &number) == 1) {
            vec.push_back(number);
        }
        fclose(file);
    }
    return vec;
}

bool write_to_file(const char *filename, const vector &vec) {

        return false;
    }
    for (size_t index = 0; index < vec.size(); index++) {
        fprintf(file, "%d\n", vec.at(index));

    fclose(file);
```

## Details

### Allocazione e deallocazione C++ style

In C++ l'allocazione e deallocazione della memoria dinamica può essere gestita in un modo più semplice rispetto a quanto si fa in C. Fino ad adesso allocavamo la memoria attraverso `realloc` e la deallocavamo attraverso `free`. Ma per creare un oggetto di tipo `vector` abbiamo utilizzato `new` e per distruggerlo abbiamo utilizzato `delete`. Ecco, in C++ è possibile utilizzare `new` e `delete` per allocare e deallocare memoria dinamica anche per i tipi di dato primitivi come `int32_t` o altri.

Concettualmente utilizzare `new` su un tipo di dato primitivo risulta essere equivalente all'utilizzarlo sul nostro tipo di dato `vector`. Ovvero, `new` alloca memoria per un oggetto di tipo `int32_t` e invoca il costruttore di default, anche se per i tipi primitivi risulta essere un'operazione superflua. In modo speculare, `delete` dealloca la memoria e invoca il distruttore di default.

Ma se volessimo creare un array dinamico di tipo `int32_t`? In questo caso possiamo ugualmente utilizzare `new` e `delete`, ma con una sintassi leggermente diversa. L'operatore `new` può essere chiamato con la sintassi `new int32_t[n]`, dove `n` è il numero di elementi da allocare. In questo caso, l'operatore `new` alloca memoria per un array di `n` elementi di tipo `int32_t` e invoca il costruttore di default per ciascun elemento dell'array. Analogamente, l'operatore `delete` può essere chiamato con la sintassi `delete[] array`, dove `array` è un puntatore all'array allocato con `new`. In questo caso, l'operatore `delete` dealloca la memoria e invoca il distruttore di default per ciascun elemento dell'array. È importante utilizzare `delete[]` in presenza di array e non il solo `delete`. Questo perché `delete` invoca il distruttore di default solo per il primo elemento dell'array, mentre `delete[]` invoca il distruttore di default per tutti gli elementi dell'array.

```cpp
// ...

struct vector {
    // ...

    ~vector() {
        delete[] data_;
    }

    void push_back(int32_t number) {
        if (size_ >= capacity_) {
            size_t temp_capacity = capacity_ == 0 ? 1 : capacity_ * 2;
            int32_t *temp_data = new int32_t[temp_capacity];
            for (size_t index = 0; index < size_; index++) {
                temp_data[index] = data_[index];
            }
            delete[] data_;
            data_ = temp_data;
            capacity_ = temp_capacity;
        }
        data_[size_] = number;
        size_++;
    }

    // ...
};

// ...
```

Notare come la funzione `push_back` sia stata modificata per utilizzare `new` e `delete[]` al posto di `realloc` e `free`. La funzione di `realloc` non può essere sostituita 1:1 con `new` e `delete[]`, perché `realloc` copia automaticamente i dati dalla vecchia area di memoria a quella nuova, mentre con `new` e `delete[]` dobbiamo fare questa copia manualmente.

### Constructor with starting capacity

Abbiamo visto che il C++ consente il method overloading, ovvero la possibilità di avere più metodi con lo stesso nome ma con parametri diversi. In questo caso possiamo creare un costruttore che accetta un parametro `capacity` per inizializzare il vettore con una capacità iniziale specificata.

```cpp
// ...

struct vector {
    // ...

    vector() {
        size_ = 0;
        capacity_ = 0;
        data_ = NULL;
    }

    vector(size_t capacity) {
        size_ = 0;
        capacity_ = capacity;
        data_ = new int32_t[capacity];
    }

    // ...
};

// ...
```

In questo modo possiamo creare un oggetto `vector` con una capacità iniziale specificata. Se viene specificata una capacità iniziale, allora non viene più utilizzato il costruttore di default, ma il costruttore con parametro `capacity`.

```cpp
int main() {
    vector vec1;                    // Default constructor
    vector vec2 = vector();         // Default constructor
    vector *vec3 = new vector;      // Default constructor
    vector *vec4 = new vector();    // Default constructor

    vector vec5(10);                // Constructor with starting capacity
    vector vec6 = vector(10);       // Constructor with starting capacity
    vector *vec7 = new vector(10);  // Constructor with starting capacity

    return 0;
}
```

Per semplificare il `main` possiamo creare una funzione di supporto che abbia lo scopo di scrivere sul file il vettore ordinato che è stato prodotto. In questo modo il `main` risulterà più lineare e pulito. Idealmente vorremmo che il `main` risultasse come segue.

```cpp
// ...
```

Quindi dovremo scrivere una funzione di supporto che permetta di passare alla funzione una stringa rappresentante il nome del file su cui scrivere e il vettore da scrivere. La funzione restituirà un valore booleano per indicare se l'operazione di scrittura è andata a buon fine o meno. La prima funzione che ci verrebbe da scrivere potrebbe apparire come il codice di seguito.

Sembra tutto quanto corretto. Parametro `const char *filename` per poter aprire in scrittura il file e il parametro `const vector vec` per poter leggere il contenuto del vettore, in più segnato come `const` per essere sicuri che la funzione non ne alteri il contenuto interno.

#### Shallow copy and double free

C'è solamente un piccolo problema se proviamo a eseguire il codice appena scritto. L'errore ci segnalerà `free(): double free detected`. Questo errore è sottile da comprendere. Iniziamo ad analizzare quanto succede. La funzione prende come parametro `const vector vec`, di conseguenza la variabile `vec` viene passata per valore e non per riferimento, come sappiamo dalla programmazione in C classica. Quello che succede se proviamo a eseguire il debugger è quanto segue.

```

├── vector vec(10);
│   │
│   └── vector(size_t capacity) {
│       └── this = 0x0000ffffe45510c8 [LABEL vec01]
│
├── bool result = write_to_file(output_filename, vec);
│   │
│   ├── vector(const vector &) {
│   │   ├── this = 0x0000ffffe45510e0 [LABEL vec02]
│   │   └── ""   = 0x0000ffffe45510c8 [LABEL vec01]
│   │
│   ├── bool write_to_file(const char *filename, const vector vec) {
│   │   └── vec = 0x0000ffffe45510e0 [LABEL vec02]
│   │
│   └── ~vector() {
│       └── this = 0x0000ffffe45510e0 [LABEL vec02]
│
└── return EXIT_SUCCESS;
    │
    └── ~vector() {

```

Nel `main` viene eseguito il suo codice tutto correttamente, tra cui anche la definizione e inizializzazione di `vec` attraverso il costruttore con capacità iniziale definito precedentemente.

Quando arriviamo a eseguire la nostra funzione appena definita `write_to_file`, vediamo che prima di entrare al suo interno viene eseguita un metodo della `struct vector` che non abbiamo definito `vector(const vector &)`. Questo metodo è generato automaticamente dal compilatore e nello specifico è il costruttore di copia. D'altronde abbiamo definito il parametro `vec` della funzione come parametro passato per valore, quindi concettualmente risulta normale che venga eseguita una copia della nostra variabile.

Quello che non risulta normale, però, è che essendo il copy constructor definito di default dal costruttore, la copia della variabile è una copia superficiale, ovvero viene copiata solamente la struttura del nostro vettore, quindi i valori `size_`, `capacity_` e `data_` per come sono definiti. Ma dato che `data_` è un puntatore a una zona di memoria, all'interno di `data_` è memorizzato un indirizzo di memoria, non il contenuto stesso di data. Di conseguenza avremo che il vettore interno alla funzione `write_to_file` e il vettore definito nel `main` avranno entrambi `data_` che punterà alla stessa zona di memoria. Questa tipologia di copia prende il nome di **shallow copy** e in questo caso è la causa del nostro problema.

Come possiamo notare dall'esecuzione del debugger, prima di eseguire la funzione `write_to_file` viene eseguita la shallow copy per il parametro `vec`. Possiamo verificare che sia effettivamente così grazie agli indirizzi di memoria in cui le variabili sono memorizzate. L'indirizzo di `this` del costruttore di copia coincide con l'indirizzo di `vec` locale alla funzione. Successivamente `write_to_file` esegue tutto il suo codice fino a restituire il risultato senza nessun errore. Ora il compilatore esegue il distruttore per la variabile locale `vec`, in quanto il suo scope era esterno alla funzione `write_to_file` dato che il costruttore di copia è stato eseguito prima della funzione stessa.

Ma analizzando cosa esegue il distruttore notiamo immediatamente il problema. Il distruttore dealloca tutta la memoria di `data_` e come abbiamo detto prima, in quanto i due vettori erano copiati tramite shallow copy, dealloca anche la memoria di data del vettore `vec` definito all'interno del `main`. Infatti concludendo l'esecuzione del programma, il compilatore prima di terminare dealloca tutte le variabili sullo stack, tra cui anche `vec`, ma dato che `data_` era già stata precedentemente deallocata, la `free` non può deallocare memoria per questa variabile. Ecco la causa del problema sollevato.

La soluzione a questo problema è molto più semplice di quanto possa sembrare. Per l'obiettivo della funzione `write_to_file` non è necessario passare il parametro `vec` per valore, basterebbe passare il parametro per riferimento.

```cpp
// ...

bool write_to_file(const char *filename, const vector *vec) {
    FILE *file = fopen(filename, "w");
    if (file == NULL) {
        return false;
    }
    for (size_t index = 0; index < vec->size(); index++) {
        fprintf(file, "%d\n", vec->at(index));
    }
    fclose(file);
    return true;
}

// ...
```

#### Parametri per riferimento

In C++ viene introdotta una sintassi nuova rispetto il C per passare i parametri per riferimento. In C sappiamo che la sintassi è `const vector *vec` e questa variabile conterrà l'indirizzo di memoria in cui il contenuto risiede. Per accedervi quindi dovremo utilizzare la sintassi dei puntatori, quindi `(*vec).size_` o semplificata `vec->size_`, ovvero è necessario de-referenziare il puntatore.

In C++ è stata introdotta una sintassi alternativa per passare i parametri per riferimento, ovvero `const vector &vec`. Questa sintassi ci permette di utilizzare `vec` come fosse una variabile locale `vec.size_` e quindi non un puntatore. Il motivo è molto semplice, `vec` anziché contenere l'indirizzo di memoria, contiene direttamente il contenuto puntato.

```
const vector *vec
    -> vec  = Indirizzo di memoria
    -> *vec = Contenuto della memoria puntata dall'indirizzo

const vector &vec
    -> &vec = Indirizzo di memoria
    ->  vec = Contenuto della memoria puntata dall'indirizzo
```

Questa sintassi permette un utilizzo dei riferimenti più semplice a livello di scrittura del codice. Quindi la funzione `write_to_file` e il codice `main` risulteranno in questo modo.

```cpp
// ...

bool write_to_file(const char *filename, const vector &vec) {
    FILE *file = fopen(filename, "w");
    if (file == NULL) {
        return false;
    }
    for (size_t index = 0; index < vec.size(); index++) {
        fprintf(file, "%d\n", vec.at(index));
    }
    fclose(file);
    return true;
}

// ...

int main(int argc, char **argv) {
    // ...

    bool result = write_to_file(output_filename, vec);
    if (result == false) {
        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

### Funzioni di supporto per lettura

Per semplificare ulteriormente il `main` possiamo aggiungere una seconda funzione di supporto per leggere i numeri dal file e restituire direttamente un vettore contenente questi numeri letti dalla funzione stessa. In prima battuta potremmo definire la funzione `read_from_file` come mostrato di seguito.

```cpp
// ...

vector read_from_file(const char *filename, size_t capacity) {
    vector vec(capacity);
    FILE *file = fopen(filename, "r");
    if (file != NULL) {
        int32_t number;
        while (fscanf(file, "%d", &number) == 1) {
            vec.push_back(number);
        }
        fclose(file);
    }
    return vec;
}

// ...
```

In modo tale da sfruttare il costruttore con una capacità iniziale ed evitare troppe estensioni della memoria `data_` e avere un `main` molto più snello e semplice da comprendere. Ma restituendo sempre un vettore come risultato abbiamo bisogno di un meccanismo che ci consenta di valutare se la lettura è avvenuta con successo. Possiamo quindi estendere `struct vector` con un metodo `empty` che permette di valutare se il vettore è pieno o meno.

```cpp
// ...

struct vector {
    // ...

    bool empty() const {
        return size_ == 0;
    }
};

// ...

int main(int argc, char **argv) {
    // ...

    vector vec = read_from_file(input_filename, 10);
    if (vec.empty() == true) {
        printf("Error: Could not open file %s\n", input_filename);
        return EXIT_FAILURE;
    }

    // ...
}
```

Provando a eseguire il codice per verificare la sua correttezza, noteremo che viene sollevato nuovamente un errore di tipo `free(): double free detected`. Come prima possiamo eseguire nuovamente il debugger per valutare cosa succede durante l'esecuzione.

Il trace del debugger mostra un comportamento particolare. Succede tutto proprio sulla riga che abbiamo modificato, ovvero `vector vec = read_from_file(input_filename, 10)`. Questo statement chiama inevitabilmente l'esecuzione della funzione `read_from_file`, la quale creerà un vettore locale `vec` attraverso il costruttore con capacità iniziale, tutto come previsto al momento.

Dopo aver completato tutto il suo codice, la funzione `read_from_file` deve restituire il vettore `vec` che ha creato e completo di tutti i numeri letti dal file. Ora inizia la parte interessante. Prima di eseguire la `return`, viene chiamato il metodo copy constructor di default come avevamo visto precedentemente. Se ci pensiamo, la signature della funzione presenta come valore di ritorno un vettore passato per valore, quindi il costruttore di copia sembra essere coerente con quanto abbiamo scritto.

La particolarità, come abbiamo già visto, è che il copy constructor di default esegue una shallow copy. Di conseguenza, al momento della distruzione del vettore temporaneo creato per essere restituito, il vettore locale viene deallocato e quindi anche la memoria `data_` del vettore temporaneo viene deallocata. Esattamente il problema visto precedentemente.

A questo punto, però, il vettore temporaneo non viene passato direttamente al vettore `vec` definito nel `main`, ma viene a sua volta creata una copia attraverso lo stesso costruttore. Questo può sembrare starno, ma è perfettamente logico. Il vettore temporaneo ha a sua volta uno scope, quindi quando termina questo scope la variabile deve essere distrutta. Per evitare che il vettore temporaneo venga distrutto, il compilatore crea una copia del vettore temporaneo e la passa al vettore `vec` definito nel `main`. Ma anche questa copia è una shallow copy, quindi il vettore `vec` definito nel `main` ha un puntatore `data_` che punta alla stessa zona di memoria del vettore temporaneo. Quando il vettore temporaneo viene distrutto, la memoria viene deallocata e quindi il vettore `vec` definito nel `main` ha un puntatore `data_` che punta a una zona di memoria già deallocata. Ecco perché viene sollevato l'errore di double free.

#### NRVO e Guaranteed Copy Elision

Il problema che abbiamo appena visto è dovuto a due concetti molto importanti in C++, il **Named Return Value Optimization**, **NRVO**, e il **Guaranteed Copy Elision**. Per mostrare questo comportamento del programma abbiamo utilizzato i seguenti flag e standard C++, in modo tale da disattivare queste ottimizzazioni.

```make
CXXFLAGS ?= -std=c++11 -fno-elide-constructors -g -Wall -Wextra -Werror -Wpedantic
```

Infatti, prima del C++17, sia il NRVO sia il Guaranteed Copy Elision erano ottimizzazioni opzionali, non definite esplicitamente nello standard. A partire dal C++17, invece, il Guaranteed Copy Elision è diventato un'ottimizzazione obbligatoria, quindi il compilatore deve sempre eseguirla, mentre il NRVO rimane un'ottimizzazione opzionale. Per questo motivo, al fine di mostrare entrambi i comportamenti, abbiamo utilizzato lo standard C++11 e disattivato le ottimizzazioni.

Entrambe le ottimizzazioni hanno lo scopo di evitare copie superflue quando un oggetto viene restituito da una funzione, ma si applicano a due situazioni diverse. Il NRVO è un'ottimizzazione che permette al compilatore di evitare la copia di una variabile locale con nome quando viene restituita da una funzione, da qui il termine Named Return Value Optimization. Il Guaranteed Copy Elision, invece, riguarda il caso in cui il valore restituito da una funzione - un'espressione senza nome, e quindi priva di un'identità propria - viene usato per inizializzare direttamente un altro oggetto dello stesso tipo. Il Guaranteed Copy Elision viene talvolta chiamato anche **Return Value Optimization**, **RVO**, in quanto differisce dal NRVO per il fatto che non richiede che l'oggetto restituito abbia un nome.

Nel nostro codice, il NRVO eviterebbe la copia della variabile locale `vec` definita nella funzione `read_from_file` verso lo spazio del valore di ritorno, mentre il Guaranteed Copy Elision eviterebbe che quello spazio, a sua volta, venga copiato in un oggetto separato per inizializzare `vec` nel `main`. In entrambi i casi il compilatore evita una copia, ma in punti diversi della catena. In parole più semplici, il NRVO eviterebbe la prima copia tramite copy constructor, mentre il Guaranteed Copy Elision eviterebbe la seconda. È importante notare che con il Guaranteed Copy Elision non si tratta semplicemente di "saltare una copia" tra due oggetti già esistenti, il vettore restituito dalla funzione non viene mai creato come oggetto a sé. Lo spazio destinato al valore di ritorno coincide, fin dall'inizio, con lo spazio di `vec` in `main`.

Il meccanismo con cui queste ottimizzazioni funzionano è abbastanza complesso, ma a grandi linee possiamo dire che permettono al compilatore di costruire l'oggetto direttamente nella sua posizione finale, evitando fin dall'inizio la creazione di un oggetto intermedio. Non si tratta quindi di copiare né di "spostare" risorse da un oggetto a un altro, perché quell'altro oggetto non arriva mai a esistere.

#### Copy constructor

Per risolvere il problema dovuto alle double free, e di conseguenza alle shallow copy, abbiamo solo una via percorribile, definire un costruttore di copia personalizzato per la nostra `struct vector`. Infatti, a differenza della funzione `write_to_file` che poteva essere modificata per passare il parametro per riferimento, la funzione `read_from_file` deve restituire un vettore per valore, quindi non possiamo modificare la sua signature. L'unica soluzione è definire un costruttore di copia personalizzato che esegua una deep copy, ovvero che copi anche il contenuto della memoria puntata da `data_`.

```cpp
// ...

struct vector {
    // ...

    vector(const vector &other) {
        size_ = other.size_;
        capacity_ = other.capacity_;
        data_ = new int32_t[capacity_];
        for (size_t index = 0; index < size_; index++) {
            data_[index] = other.data_[index];
        }
    }

    // ...
}

// ...
```

Grazie a questa sola modifica, il nostro programma precedentemente descritto sarà in grado di funzionare correttamente. Il costruttore di copia richiede come parametro un riferimento a un oggetto `vector` esistente, e crea un nuovo oggetto `vector` che è una copia profonda dell'oggetto originale. In questo modo, il nuovo oggetto avrà la propria copia dei dati, evitando problemi di double free. Di seguito possiamo verificare la sua correttezza anche attraverso il trace del debugger.

La causa è un altro metodo non definito da noi, di conseguenza un metodo di default e quindi svolto attraverso shallow copy. Questo metodo definito dal compilatore prende il nome di *assignment operator* e lo approfondiremo tra poco. Il motivo per cui non viene mostrato nessun errore durante l'esecuzione è sottile. Il rilevamento del double free avviene a runtime, ad opera dell'allocatore di memoria, la parte della libreria standard del C che gestisce l'heap. Quando una zona di memoria viene liberata, l'allocatore vi scrive alcuni metadati interni per tenerne traccia. Nel nostro caso, tra la prima free, quella eseguita alla distruzione del vettore temporaneo restituito da `read_from_file`, e la seconda, quella eseguita a fine `main`, il vettore `vec` possiede ancora quello stesso puntatore, ormai dangling, e chiamiamo `vec.sort()` proprio su di esso. Ecco perché l'errore non emerge più, non è che il problema sia sparito, è solo diventato invisibile.

Proviamo ora a capire perché con una piccolissima variazione del codice in `main` viene alterata l'esecuzione rispetto a quanto visto fino a questo punto. Il motivo riguarda al contesto in cui viene eseguito l'operatore `=`.

Infatti l'operatore `=` posto al fianco della definizione di una variabile fa sì che venga eseguita l'inizializzazione della variabile stessa, mentre l'operatore `=` posto in un contesto successivo alla definizione della variabile fa sì che venga eseguito l'assegnamento del valore a una variabile già inizializzata. Volendo fare un esempio molto più semplice per comprenderne la differenza:

```cpp
int main() {
    int a = 10; // Inizializzazione
    a = 20;     // Assegnamento
    return 0;
}
```

Per un tipo primitivo come `int`, la differenza tra inizializzazione e assegnamento è quasi impercettibile. In entrambi i casi il valore viene semplicemente copiato. Ma per un tipo classe come il nostro `vector`, come abbiamo appena visto nei due trace del debugger, la differenza è sostanziale, l'inizializzazione invoca il costruttore di copia, l'assegnamento invoca l'operatore di assegnamento. Sono due metodi distinti del tutto, con implementazioni distinte, ed è proprio per questo che avendo corretto solo il primo, il nostro programma torna a rompersi non appena passiamo dall'inizializzazione all'assegnamento.

Volendo essere del tutto trasparenti, il comportamento undefined appena visto sull'output deriva dal fatto che abbiamo compilato utilizzando un ulteriore flag, quello `-Wno-deprecated-copy`. Questo errore del compilatore che abbiamo soppresso ci avvisava che l'operatore di assegnamento di default generato dal compilatore è deprecato, quindi non dovrebbe essere utilizzato. Questo perché l'operatore di assegnamento di default esegue una shallow copy, quindi può causare problemi di double free come abbiamo visto.

In generale, in C++ si segue la regola del [Rule of Three](https://en.wikipedia.org/wiki/Rule_of_three_(C++_programming)), che afferma che se una classe definisce un distruttore, un costruttore di copia o un operatore di assegnamento, allora dovrebbe definire anche gli altri due. Questo perché la gestione della memoria dinamica richiede una gestione coerente delle risorse, e se uno di questi metodi non è definito correttamente, può portare a problemi come quelli che abbiamo visto. Questa regola ci sarebbe stata utile all'inizio di questa versione per scrivere correttamente la nostra `struct vector`, ma abbiamo preferito introdurre i concetti passo passo per comprendere meglio il funzionamento del C++.

#### Assignment operator

Quindi per completare il nostro programma al fine di avere un comportamento corretto, dobbiamo definire un operatore di assegnamento personalizzato per la nostra `struct vector`. Questo operatore deve eseguire una deep copy dei dati, in modo simile a quanto fatto nel costruttore di copia.

```cpp
// ...

struct vector {
    // ...

    vector &operator=(const vector &rhs) {
        if (this != &rhs) {
            size_ = rhs.size_;
            capacity_ = rhs.capacity_;
            delete[] data_;
            data_ = new int32_t[capacity_];
            for (size_t index = 0; index < size_; index++) {
                data_[index] = rhs.data_[index];
            }
        }
        return *this;
    }

    // ...
}

// ...
```

L'operatore di assegnamento richiede un riferimento a un oggetto `vector` esistente come parametro, e copia i dati da quell'oggetto al nuovo oggetto. Prima di copiare i dati, dealloca la memoria esistente per evitare **memory leak**. Infine, restituisce un riferimento a se stesso per consentire l'assegnamento concatenato. Il nome `rhs` sta per **Right Hand Side**, ovvero il lato destro dell'operatore di assegnamento, mentre `*this` rappresenta l'oggetto a sinistra dell'operatore di assegnamento. La sintassi `return *this` restituisce un riferimento all'oggetto corrente, permettendo così di concatenare più operazioni di assegnamento in una sola riga di codice. Questa sintassi è essenziale al fine di permettere l'assegnamento concatenato:

```cpp
int main() {
    vector vec1, vec2, vec3;
    vec1 = vec2 = vec3; // Assegnamento concatenato
    return 0;
}
```
 Infatti, ogni operatore di assegnamento restituisce un riferimento a se stesso, quindi l'operatore di assegnamento successivo può essere chiamato sull'oggetto restituito. In questo modo, possiamo concatenare più operazioni di assegnamento in una sola riga di codice. Il controllo `if (this != &rhs)` è necessario per gestire correttamente il caso dell'auto-assegnamento, ovvero quando un oggetto viene assegnato a se stesso, ad esempio `vec = vec`. Senza questo controllo, l'istruzione `delete[] data_` deallocherebbe la memoria prima ancora di copiarne il contenuto in `rhs.data_`, ma se `rhs` è lo stesso oggetto di `*this`, quella memoria sarebbe proprio quella da cui stiamo cercando di leggere, causando dati corrotti.

Attenzione, il metodo `operator=` è sì un metodo particolare, ma nessuno vieta di chiamarlo direttamente come un metodo della classe, quindi `vec1.operator=(vec2)` è perfettamente corretto e funziona esattamente come `vec1 = vec2`. La sintassi dell'operatore di assegnamento è solo una convenzione per rendere il codice più leggibile e intuitivo.

Quindi definito sia il **copy constructor** che l'**assignment operator** siamo stati in grado di risolvere tutti i problemi di double free e memory leak presenti nel nostro codice. Il nostro programma ora è completo e funzionante, e possiamo eseguire il sorting dei numeri letti da un file e scrivere il risultato in un altro file senza problemi.

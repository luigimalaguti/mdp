namespace mdp {
    template <typename T>
    int compare(const void *pointer_a, const void *pointer_b) {
        T value_a = *(T *)pointer_a;
        T value_b = *(T *)pointer_b;
        return (value_a > value_b) - (value_a < value_b);
    }

    template <typename T>
    class vector {
    private:
        size_t size_;
        size_t capacity_;
        T *data_;

    public:
        vector(size_t capacity = 0) : size_(0), capacity_(capacity), data_(nullptr) {
            if (capacity_ > 0) {
                data_ = new T[capacity_];
            }
        }

        vector(const vector &other) : size_(other.size_), capacity_(other.capacity_), data_(new T[capacity_]) {
            std::copy_n(other.data_, size_, data_);
        }

        vector(vector &&other) : size_(other.size_), capacity_(other.capacity_), data_(other.data_) {
            other.data_ = nullptr;
        }

        ~vector() {
            delete[] data_;
        }

        vector &operator=(vector rhs) {
            swap(*this, rhs);
            return *this;
        }

        const T &operator[](size_t index) const {
            return data_[index];
        }

        T &operator[](size_t index) {
            return data_[index];
        }

        void push_back(const T &number) {
            if (size_ >= capacity_) {
                size_t temp_capacity = capacity_ == 0 ? 1 : capacity_ * 2;
                T *temp_data = new T[temp_capacity];
                std::copy_n(data_, size_, temp_data);
                delete[] data_;
                data_ = temp_data;
                capacity_ = temp_capacity;
            }
            data_[size_] = number;
            size_++;
        }

        void sort() {
            qsort(data_, size_, sizeof(T), compare<T>);
        }

        size_t size() const {
            return size_;
        }

        const T &at(size_t index) const {
            assert(index < size_);
            return data_[index];
        }

        bool empty() const {
            return size_ == 0;
        }

        friend void swap(vector &vector_a, vector &vector_b) {
            using std::swap;
            swap(vector_a.size_, vector_b.size_);
            swap(vector_a.capacity_, vector_b.capacity_);
            swap(vector_a.data_, vector_b.data_);
        }
    };

### Standard vector library

In questa ultima versione del nostro codice riguardo l'esercizio *sort int* ci focalizziamo nel migliorare il codice utilizzando le tecniche avanzate e best practices del C++.

Quando si vuole scrivere del codice che possa uscire dallo scopo didattico o di esercizio, è importante utilizzare le librerie standard del C++ come la standard library, in quanto queste librerie sono state sviluppate e testate da esperti del settore e garantiscono una maggiore efficienza e sicurezza rispetto a soluzioni personalizzate. Inoltre sono talmente vaste e complete che spesso è possibile trovare già implementate le funzionalità di cui si ha bisogno, evitando di dover reinventare la ruota.

Proprio per questo motivo la prima modifica che facciamo al nostro codice è rimuovere completamente tutta la classe `vector` e namespace `mdp` che abbiamo sviluppato fino a questo punto. Lo scopo di sviluppare questa classe, infatti, non era quello di avere una soluzione ottimale e pronta per applicazioni differenti, ma piuttosto comprendere le principali caratteristiche del linguaggio C++.

### Iterators and range-based for

Per provare ad alzare il livello della qualità del nostro codice possiamo concentrarci sulla funzione di supporto `write_to_file`.

Questa funzione per scrivere il contenuto del nostro vettore sul file svolge un ciclo `for` su tutto il contenuto di esso, a prescindere dalla sua dimensione. Accedere attraverso un indice può risultare dispendioso in termini di performance, soprattutto se il vettore è molto grande. Più la dimensione del vettore aumenta, più il tempo di accesso agli elementi attraverso un indice diventa lungo. Inoltre, accedere attraverso un indice presuppone che il contenitore in cui sono memorizzati gli elementi sia un contenitore indicizzato, come ad esempio un array o un vettore. In caso contrario, l'accesso agli elementi attraverso un indice non è possibile. Se dovessimo per qualche motivo cambiare il contenitore in cui sono memorizzati gli elementi, dovremmo riscrivere la funzione `write_to_file` per adattarla al nuovo contenitore.

Per questo motivo esistono costrutti del linguaggio C++ che permettono di accedere agli elementi di un contenitore in maniera più efficiente. Uno di questi costrutti è l'**iteratore**, il quale permette di accedere agli elementi di un contenitore senza dover utilizzare un indice.

```cpp
// ...

bool write_to_file(const char *filename, const std::vector<int32_t> &vec) {
    std::ofstream os(filename);
    if (!os) {
        return false;
    }
    std::vector<int32_t>::const_iterator start = vec.begin();
    std::vector<int32_t>::const_iterator stop = vec.end();
    std::vector<int32_t>::const_iterator it;
    for (it = start; it != stop; it++) {
        os << *it << "\n";
    }
    return true;
}

// ...
```

Questo codice mostra come si presenta un iteratore. Ogni contenitore della libreria standard del C++ espone i metodi `begin()` e `end()`. Questi due metodi restituiscono un valore di tipo `std::vector<int32_t>::const_iterator`, dove `std::vector<int32_t>` è il tipo del contenitore e `const_iterator` è il tipo dell'iteratore.

L'iteratore in sé è un oggetto che permette di accedere agli elementi del contenitore. Essenzialmente l'iteratore è un puntatore che punta all'elemento corrente del contenitore. È quindi un puntatore a una zona di memoria in cui sono memorizzati gli elementi del contenitore. L'iteratore può essere incrementato o decrementato per accedere agli elementi successivi o precedenti del contenitore.

Quindi il ciclo `for`, ma eventualmente anche `while` dato che sono analoghi, assegna all'iteratore `it` il valore restituito da `begin()`, che è l'indirizzo del primo elemento del contenitore. Il ciclo continua fino a quando l'iteratore `it` non raggiunge l'indirizzo restituito da `end()`, che è l'indirizzo successivo all'ultimo elemento del contenitore.

È molto importante utilizzare la condizione `it != stop` e non `it < stop`. Infatti ogni contenitore della libreria standard del C++ può avere un tipo di iteratore diverso. Alcuni contenitori memorizzano gli elementi in maniera contigua, come ad esempio `std::vector`, mentre altri contenitori non lo fanno, come ad esempio `std::list`. In questi casi l'operatore `<` non è definito per gli iteratori, quindi non è possibile utilizzarlo. L'operatore `!=` invece è definito per tutti i tipi di iteratore, quindi è possibile utilizzarlo in tutti i casi.

Per accedere al valore puntato dall'oggetto iteratore `it` è necessario utilizzare l'operatore di dereferenziazione `*`. Questo operatore restituisce il valore dell'elemento puntato dall'iteratore. Nota importante, non dereferenziare l'iteratore `stop`, in quanto esso non punta a un elemento valido del contenitore, ma all'indirizzo successivo all'ultimo elemento del contenitore, causando un *undefined behavior*.

Questo metodo di accesso al vettore non è tanto più performante rispetto all'indice, in quanto per `std::vector` è esattamente la stessa cosa. Ma questo meccanismo ci consente di avere molta più flessibilità nel caso in cui dovessimo cambiare il contenitore in cui sono memorizzati gli elementi che non supporta l'indicizzazione.

Sembra quindi un costrutto molto efficace, ma brutto e complesso da scrivere. Quindi si potrebbe provare a compattare un minimo la sintassi del `for`, per fare in modo che ci sia meno codice da scrivere e leggere.

```cpp
// ...

bool write_to_file(const char *filename, const std::vector<int32_t> &vec) {
    std::ofstream os(filename);
    if (!os) {
        return false;
    }
    for (std::vector<int32_t>::const_iterator it = vec.begin(); it != vec.end(); it++) {
        os << *it << "\n";
    }
    return true;
}

// ...
```

È possibile evitare di definire l'iteratore di inizio e fine del contenitore, e definire direttamente l'iteratore `it` all'interno della condizione del ciclo `for`. In questo modo il codice risulta più compatto.

Ma anche in questo modo il codice non risulta molto bello e leggibile. Infatti la sintassi del ciclo `for` è molto complessa e difficile da leggere, soprattutto per chi non ha molta esperienza con il linguaggio C++.

Per questo motivo, il linguaggio C++ ha introdotto una keyword particolare chiamata `auto`. Questa keyword viene utilizzata per specificare il tipo di dato di una variabile, il quale viene dedotto automaticamente dal compilatore in base al valore assegnato alla variabile stessa. In questo modo il codice risulta più leggibile e meno complesso.

```cpp
// ...

bool write_to_file(const char *filename, const std::vector<int32_t> &vec) {
    std::ofstream os(filename);
    if (!os) {
        return false;
    }
    for (auto it = vec.begin(); it != vec.end(); it++) {
        os << *it << "\n";
    }
    return true;
}

// ...
```

In questo modo il codice ha subito un miglioramento notevole in termini di leggibilità e semplicità. Infatti il tipo dell'iteratore viene dedotto automaticamente dal compilatore, quindi non è più necessario scrivere il tipo completo dell'iteratore.

Ma anche con questa sintassi il codice potrebbe essere migliorato. Infatti, quando si utilizzano elementi del contenitore che risultano essere più complessi di un semplice tipo primitivo, l'utilizzo della dereferenziazione dell'iteratore può risultare poco leggibile e difficile da comprendere. Per questo motivo, gli sviluppatori nel tempo hanno iniziato a scrivere i propri `for` in questo modo.

```cpp
// ...

bool write_to_file(const char *filename, const std::vector<int32_t> &vec) {
    std::ofstream os(filename);
    if (!os) {
        return false;
    }
    for (auto it = vec.begin(); it != vec.end(); it++) {
        const auto &value = *it;
        os << value << "\n";
    }
    return true;
}

// ...
```

Viene definita una variabile di supporto `value` costante, perché non è necessario modificare il valore dell'elemento del contenitore, e viene assegnato il valore dell'elemento puntato dall'iteratore `it`. In questo modo il codice risulta più leggibile e comprensibile, in quanto è chiaro che `value` rappresenta l'elemento corrente del contenitore. Il tipo di dato che viene dedotto da `auto` non prende in considerazione il fatto che l'elemento possa essere un riferimento o un valore, permettendo maggiore flessibilità. Per questo motivo dobbiamo specificare che `value` è un riferimento costante, in modo da evitare di copiare il valore dell'elemento del contenitore, ma di accedere direttamente al valore originale.

Seppur sia stata semplificata la scrittura del `for`, l'utilizzo di un alias o di un puntatore per accedere al valore dell'elemento del contenitore può risultare poco leggibile e difficile da comprendere. Per questo motivo, il linguaggio C++ ha introdotto una nuova sintassi per il ciclo `for`, chiamata **range-based for**.

```cpp
// ...

bool write_to_file(const char *filename, const std::vector<int32_t> &vec) {
    std::ofstream os(filename);
    if (!os) {
        return false;
    }
    for (const auto &value : vec) {
        os << value << "\n";
    }
    return true;
}

// ...
```

Questo costrutto permette di iterare direttamente sugli elementi del contenitore, senza dover utilizzare un iteratore o un indice. La sintassi è molto semplice e leggibile, in quanto è chiaro che `value` rappresenta l'elemento corrente del contenitore `vec`. La sintassi generica, quindi, è essenzialmente `for (range_declaration : range_expression) statement`, dove `range_declaration` è la dichiarazione della variabile che rappresenta l'elemento corrente del contenitore, `range_expression` è l'espressione che rappresenta il contenitore da iterare e `statement` è il blocco di codice da eseguire per ogni elemento del contenitore.

Con il costrutto **range-based for** non è possibile accedere all'indice dell'elemento corrente del contenitore, di conseguenza se si ha bisogno di accedere all'indice dell'elemento corrente del contenitore, è necessario utilizzare un iteratore o un indice. L'utilizzo di un costrutto o di un altro dipende dalle esigenze del codice.

Adesso passiamo ad alzare il livello della qualità del nostro codice analizzando la funzione di supporto `read_from_file`. La funzione atturalmente si presenta in questo modo.

```cpp

    if (is) {
        int32_t number;

```

Il linguaggio C++, come detto, mette a disposizione molte librerie standard, le quali possono avere un ampio spettro di funzionalità. La libreria `iterator` mette a disposizione una funzione chiamata `std::istream_iterator`, la quale permette di leggere direttamente gli elementi da un flusso di input, come ad esempio un file, senza dover utilizzare funzioni sullo stream come `operator>>` o `get()`. La funzione `std::istream_iterator` è un iteratore di input, il quale permette di leggere gli elementi da un flusso di input in maniera sequenziale.

```cpp
// ...

std::vector<int32_t> read_from_file(const char *filename) {
    std::vector<int32_t> vec;
    std::ifstream is(filename);
    if (is) {
        std::istream_iterator<int32_t> start = is;
        std::istream_iterator<int32_t> stop;
        for (auto it = start; it != stop; it++) {
            const auto &value = *it;
            vec.push_back(value);
        }
    }
    return vec;
}

// ...
```

Specificando il tipo di dato `int32_t` l'iteratore `istream` è in grado di leggere direttamente gli elementi dal flusso di input `is` e leggere i dati già convertiti nel tipo di dato specificato. Concettualmente il costrutto è molto simile a quanto abbiamo fatto per la funzione di supporto `write_to_file`.

Eventualmente possiamo provare a ottimizzare il codice evitando alcune delle variabili di supporto, come ad esempio `start` e `stop`, e definendo direttamente l'iteratore `it` all'interno della condizione del ciclo `for`.

```cpp
// ...

std::vector<int32_t> read_from_file(const char *filename) {
    std::vector<int32_t> vec;
    std::ifstream is(filename);
    if (is) {
        for (std::istream_iterator<int32_t> it = is; it != std::istream_iterator<int32_t>(); it++) {
            const auto &value = *it;
            vec.push_back(value);
        }
    }
    return vec;
}

// ...
```

Anche se questa volta non appare troppo leggibile, il codice risulta più compatto e meno dispersivo. Però possiamo unire l'utilizzo della libreria standard `vector` con la libreria standard `iterator`, in modo da rendere il codice più leggibile e compatto.

Infatti, la classe `vector` come altri container definiscono un svariato numero di costruttori, tra cui un costruttore che accetta due iteratori come parametri. Questi due iteratori possono essere semplicemente derivanti da un container di un altro tipo, come ad esempio un array o un altro vettore, oppure possono essere derivanti da un iteratore di input, come ad esempio `std::istream_iterator`. In questo caso il costruttore del vettore legge direttamente gli elementi dal flusso di input e li memorizza nel vettore.

```cpp
// ...

std::vector<int32_t> read_from_file(const char *filename) {
    std::ifstream is(filename);
    if (!is) {
        return std::vector<int32_t>();
    }
    std::istream_iterator<int32_t> start = is;
    std::istream_iterator<int32_t> stop;
    std::vector<int32_t> vec(start, stop);
    return vec;
}

// ...
```

Utilizzando queste due librerie standard insieme siamo riusciti a ottenere un codice molto più compatto, leggibile ed elegante, senza fare alcun uso di un qualsiasi tipo di ciclo. Ma possiamo provare a fare ancora di più.

```cpp
// ...

std::vector<int32_t> read_from_file(const char *filename) {
    std::ifstream is(filename);
    if (!is) {
        return std::vector<int32_t>();
    }
    std::vector<int32_t> vec(std::istream_iterator<int32_t>(is), std::istream_iterator<int32_t>());
    return vec;
}

// ...
```

Anziché definire due variabili di supporto `start` e `stop`, possiamo definire direttamente gli iteratori all'interno della chiamata al costruttore del vettore. La variabile `start` viene sostituita con un oggetto temporaneo `std::istream_iterator<int32_t>(is)` inizializzato con `is`, mentre la variabile `end` viene semplicemente sostituita con un oggetto temporaneo senza parametri.

C'è solo un piccolo problema, il compilatore ci segnala questo errore `warning: parentheses were disambiguated as a function declaration`. La causa di questo errore deriva dal fatto che il compilatore interpreta la riga di codice come una dichiarazione di funzione, anziché come una chiamata al costruttore del vettore.

Infatti la riga che abbiamo scritto può essere interpretata come una dichiarazione di una funzione `vec` che accetta due parametri, il primo di tipo `std::istream_iterator<int32_t>` e il secondo di tipo `std::istream_iterator<int32_t>`, e restituisce un oggetto di tipo `std::vector<int32_t>`. Per risolvere questo problema possiamo utilizzare le parentesi graffe `{}` al posto delle parentesi tonde `()`, in quanto le parentesi graffe vengono interpretate come una lista di inizializzazione, mentre le parentesi tonde vengono interpretate come una chiamata al costruttore.

```cpp
// ...

std::vector<int32_t> read_from_file(const char *filename) {
    std::ifstream is(filename);
    if (!is) {
        return std::vector<int32_t>();
    }
    std::vector<int32_t> vec{std::istream_iterator<int32_t>(is), std::istream_iterator<int32_t>()};
    return vec;
}

// ...
```

Questa è un'altra possibilità che il C++ mette a disposizione per inizializzare un oggetto, la quale è molto utile quando si vogliono inizializzare gli elementi di un contenitore con una lista di valori. In questo caso, il costruttore del vettore accetta due iteratori come parametri, i quali vengono utilizzati per inizializzare gli elementi del vettore.

Questo costrutto è molto compatto, anche se quello di seguito risulta più lungo è allo stesso tempo più leggibile e comprensibile, in quanto è chiaro che `vec` rappresenta il vettore di interi letti dal file.

```cpp
// ...

std::vector<int32_t> read_from_file(const char *filename) {
    std::ifstream is(filename);
    if (!is) {
        return std::vector<int32_t>();
    }
    std::istream_iterator<int32_t> start(is);
    std::istream_iterator<int32_t> stop;
    return std::vector<int32_t>(start, stop);
}

// ...
```

### Algorithms and predicates

Passiamo ora ad analizzare la riga di codice che abbiamo tralasciato inizialmente, il sorting del vettore all'interno del `main`. Avevamo detto che la funzione `sort` non è disponibile nella classe `std::vector`, ma è disponibile nella libreria standard del C++ `algorithm`. Questa libreria mette a disposizione un ampio spettro di algoritmi, tra cui il sorting, il quale può essere utilizzato per ordinare gli elementi di un contenitore.

```cpp
// ...

int main(int argc, char **argv) {
    // ...

    std::sort(vec.begin(), vec.end());

    // ...
}
```

La funzione `std::sort` richiede appunto due parametri di tipo iterator. Il primo parametro rappresenta l'iteratore di inizio del contenitore da ordinare, mentre il secondo parametro rappresenta l'iteratore di fine del contenitore da ordinare. In questo caso, gli iteratori vengono ottenuti tramite i metodi `begin()` e `end()` del vettore `vec` come abbiamo fatto in precedenza per la funzione di supporto `write_to_file`.

Questo è il sorting più semplice che possiamo fare, ordina tutti gli elementi del contenitore in ordine crescente. Ma la funzione `std::sort` mette a disposizione più overload, tra cui uno che accetta un terzo parametro, il quale rappresenta una funzione di confronto.

Questa funzione di confronto viene utilizzata per confrontare due elementi del contenitore e determinare l'ordine in cui devono essere ordinati. Questo terzo parametro prende il nome di **predicato**, il quale è una funzione che restituisce un valore booleano, indicando se il primo elemento deve essere ordinato prima del secondo elemento.

```cpp
// ...

int main(int argc, char **argv) {
    // ...

    std::sort(vec.begin(), vec.end(), std::greater<int32_t>());

    // ...
}
```

La funzione `std::greater` è un predicato predefinito nella libreria standard del C++, il quale confronta due elementi e restituisce `true` se il primo elemento è maggiore del secondo elemento, altrimenti restituisce `false`. In questo caso, la funzione `std::sort` ordina gli elementi del contenitore in ordine decrescente. `std::less` è il predicato opposto di `std::greater`, il quale ordina gli elementi del contenitore in ordine crescente.

La libreria `algorithm` mette a disposizione anche una funzione che potrebbe tornare utile alla funzione di supporto `read_from_file` vista poco fa, la permette di fare uso degli iterator per popolare il vettore che si vuole restituire. La funzione si presentava in questo modo.

```cpp
// ...

std::vector<int32_t> read_from_file(const char *filename) {
    std::ifstream is(filename);
    if (!is) {
        return std::vector<int32_t>();
    }
    std::istream_iterator<int32_t> start(is);
    std::istream_iterator<int32_t> stop;
    return std::vector<int32_t>(start, stop);
}

// ...
```

Grazie alla funzione `std::copy`, la versione di `std::copy_n` usata in una versione precedente, possiamo passare l'iterator di inizio e di fine del nostro stream iterator di input `std::istream_iterator` e passare un terzo parametro. Il terzo parametro è anch'esso un iterator, ma di output, il quale rappresenta l'iteratore di inizio del contenitore in cui vogliamo copiare gli elementi letti dal flusso di input. Però non possiamo passare direttamente `vec.begin()`, in quanto la funzione `std::copy` copia brutalmente gli elementi senza preoccuparsi di allocare la memoria, motivo per cui utilizzavamo `push_back()`.

Quindi ci viene in aiuto la funzione `std::back_inserter`, la quale prende come parametro un contenitore e restituisce un iteratore di output, il quale rappresenta l'iteratore di inizio del contenitore in cui vogliamo copiare gli elementi letti dal flusso di input. L'iteratore di output restituito da `std::back_inserter` utilizza il metodo `push_back()` del contenitore per aggiungere gli elementi al contenitore, quindi non è necessario preoccuparsi di allocare la memoria.

```cpp
// ...

std::vector<int32_t> read_from_file(const char *filename) {
    std::vector<int32_t> vec;
    std::ifstream is(filename);
    if (!is) {
        return vec;
    }
    std::istream_iterator<int32_t> start(is);
    std::istream_iterator<int32_t> stop;
    std::copy(start, stop, std::back_inserter(vec));
    return vec;
}

// ...
```

La funzione `std::copy` può eventualmente essere utilizzata anche per svolgere il compito opposto, ovvero scrivere gli elementi di un contenitore su un flusso di output, come ad esempio un file. Basterà passare questa volta gli iterator di inizio e fine del vettore e passare come terzo argomento l'analogo di `std::istream_iterator`, ovvero `std::ostream_iterator`. L'iterator di inizio di `std::ostream_iterator` richiede un secondo parametro oltre al flusso di output, il quale rappresenta il delimitatore da utilizzare tra gli elementi del contenitore. In questo caso utilizziamo il carattere di nuova linea `\n` come delimitatore.

```cpp
int main() {
    std::vector<int32_t> vec = {1, 2, 3, 4, 5};

    std::ofstream os("filename.txt");
    if (!os) {
        return 1;
    }

    std::ostream_iterator<int32_t> start(os, "\n");
    std::copy(vec.begin(), vec.end(), start);

    return 0;
}
```

### Object functions and lambda expressions

Tornando nuovamente ad analizzare la funzione di sorting nel codice `main` possiamo fare qualcosa di meglio e più flessibile di quanto abbiamo fatto fino ad adesso.

```cpp
// ...

int main(int argc, char **argv) {
    // ...

    std::sort(vec.begin(), vec.end(), std::less<int32_t>());

    // ...
}
```

Abbiamo capito come il terzo parametro della funzione `std::sort` sia un predicato, il quale può essere una funzione predefinita come `std::less` o `std::greater`, ma nessuno vieta di poter definire una propria funzione di confronto. Sappiamo che un predicato è una funzione che restituisce un valore booleano, quindi possiamo definire una funzione che accetta due parametri dello stesso tipo del contenitore e restituisce `true` se il primo parametro deve essere ordinato prima del secondo parametro, altrimenti restituisce `false`.

```cpp
// ...

bool sort_vector(int32_t number_a, int32_t number_b) {
    return number_a < number_b;
}

int main(int argc, char **argv) {
    // ...

    std::sort(vec.begin(), vec.end(), sort_vector);

    // ...
}
```

Questo codice funziona perfettamente e ci da molta flessibilità. Ma proviamo a mescolare un pò le carte. Non vogliamo più ordinare secondo l'ordine crescente o decrescente degli elementi, ma piuttosto secondo il valore medio dell'intero vettore. Quindi se un elemento è minore della media, deve essere ordinato prima degli elementi maggiori della media.

Quindi come prima cosa possiamo nuovamente fare uso della standard library C++ per calcolare il valore medio del vettore, facendo uso della funzione `std::accumulate` della libreria `numeric`. Questa funzione accetta tre parametri, il primo e il secondo parametro sono gli iterator di inizio e fine del contenitore, mentre il terzo parametro è il valore iniziale dell'accumulatore. La funzione restituisce la somma di tutti gli elementi del contenitore, partendo dal valore iniziale dell'accumulatore.

```cpp
// ...

bool sort_vector(int32_t number_a, int32_t number_b) {
    return number_a < number_b;
}

int main(int argc, char **argv) {
    // ...

    double sum = std::accumulate(vec.begin(), vec.end(), 0.0);
    double average = sum / (double)vec.size();
    std::sort(vec.begin(), vec.end(), sort_vector);

    // ...
}
```

Ora sorge spontanea una domanda, come facciamo a passare il valore medio del vettore alla funzione di confronto `sort_vector`? La funzione di confronto accetta solo due parametri, quindi non possiamo passare il valore medio del vettore come terzo parametro. In C++ esiste un costrutto chiamato **object function**, il quale permette di definire una funzione come oggetto, in modo da poter passare dei parametri al momento della creazione dell'oggetto. Questo costrutto è molto utile quando si vuole passare dei parametri a una funzione di confronto, come nel nostro caso. Quindi sostituiamo la funzione `sort_vector` con il seguente codice.

```cpp
// ...

class sort_vector {
    double average_;

public:
    sort_vector(double average) : average_(average) {}

    bool operator()(int32_t number_a, int32_t number_b) {
        auto distance_a = std::abs(number_a - average_);
        auto distance_b = std::abs(number_b - average_);
        return distance_a < distance_b;
    }
};

int main(int argc, char **argv) {
    // ...

    double sum = std::accumulate(vec.begin(), vec.end(), 0.0);
    double average = sum / (double)vec.size();
    std::sort(vec.begin(), vec.end(), sort_vector(average));

    // ...
}
```

Il concetto dell'object function è molto semplice. Definiamo una classe, che sia essa con `class` o `struct`, per fare in modo di poter inizializzare dei parametri al momento della creazione dell'oggetto. I parametri passati al costruttore della classe vengono memorizzati come variabili membro della classe stessa. In questo modo l'oggetto possiede dati ulteriori rispetto a quelli che vengono utilizzati come parametri della funzione di confronto. Dopodiché si fa l'overload dell'operatore `()`, il quale permette di utilizzare l'oggetto come se fosse una funzione. L'operatore `()` accetta due parametri, i quali rappresentano gli elementi del contenitore da confrontare, e restituisce un valore booleano, indicando se il primo elemento deve essere ordinato prima del secondo elemento.

Ma in C++ esiste un costrutto ancora più interessante rispetto all'object function. Questo costrutto prende il nome di **lambda expression** e permette di definire una funzione anonima, la quale può essere utilizzata come predicato. La lambda expression può essere considerata come una funzione lambda in Python o una arrow function in JavaScript. La sintassi delle lambda expression è semplice, è formata da questo insieme di simboli `[](){}`. La parte tra le parentesi quadre `[]` rappresenta la cattura delle variabili, la parte tra le parentesi tonde `()` rappresenta i parametri della funzione, mentre la parte tra le parentesi graffe `{}` rappresenta il corpo della funzione.

```cpp
// ...

int main(int argc, char **argv) {
    // ...

    double sum = std::accumulate(vec.begin(), vec.end(), 0.0);
    double average = sum / (double)vec.size();
    std::sort(vec.begin(), vec.end(), [&average](int32_t number_a, int32_t number_b) {
        auto distance_a = std::abs(number_a - average);
        auto distance_b = std::abs(number_b - average);
        return distance_a < distance_b;
    });

    // ...
}
```

Ad esclusione della parte iniziale della sintassi `[]`, la lambda expression risulta sintatticamente identica a come si presenterebbe una qualsiasi funzione in C++. La novità riguarda appunto la parte di closure, la quale permette di catturare le variabili definite al di fuori della lambda expression. Infatti di default una lambda expression non può accedere a nessuna definita esternamente al proprio body. Quindi per utilizzare queste variabili esterne, come `average`, è necessario catturarle all'interno della lambda expression. La cattura può essere fatta per valore o per riferimento, a seconda delle esigenze. In questo caso abbiamo catturato la variabile `average` per riferimento, in quanto vogliamo utilizzare il valore originale della variabile e non una copia di esso.

La parte di closure può accettare differenti pattern. Indicando la closure con `[]` non viene catturata nessuna variabile esterna, indicando la closure con `[=]` vengono catturate tutte le variabili esterne per valore, mentre indicando la closure con `[&]` vengono catturate tutte le variabili esterne per riferimento. È possibile anche catturare solo alcune variabili esterne, come nel nostro caso, indicando la closure con `[&average]`, in questo modo viene catturata solo la variabile `average` per riferimento. In alternativa si possono catturare più variabili esterne, come ad esempio `[&average, &sum]`, in questo modo vengono catturate le variabili `average` e `sum` per riferimento.

### Ranges and projections

L'ultimo concetto avanzato sul linguaggio C++ che analizzeremo riguarda una libreria standard dal nome `std::ranges`. Questa libreria rende disponibili molte funzionalità interessanti, tra cui la possibilità di utilizzare il concetto di **projection**.

La projection è un concetto che permette di trasformare gli elementi di un contenitore prima di passarli a una funzione di confronto. Essenzialmente è una funzione che riceve in input un elemento del contenitore e restituisce un valore che rappresenta l'elemento trasformato. Questo elemento trasformato viene poi utilizzato da una seconda funzione, una funzione di confronto, per confrontare gli elementi del contenitore.

Prendiamo come codice di partenza per studiarne il comportamento lo stesso codice analizzato precedentemente per il sorting del vettore in base alla distanza dalla media.

```cpp
// ...

int main(int argc, char **argv) {
    // ...

    double sum = std::accumulate(vec.begin(), vec.end(), 0.0);
    double average = sum / (double)vec.size();
    std::sort(vec.begin(), vec.end(), [&average](int32_t number_a, int32_t number_b) {
        auto distance_a = std::abs(number_a - average);
        auto distance_b = std::abs(number_b - average);
        return distance_a < distance_b;
    });

    // ...
}
```

Questo codice funziona correttamente e risulta anche molto leggibile, ma possiamo fare qualcosa di meglio. Notiamo immediatamente come la funzione `std::abs(number - average)` viene applicata a entrambi gli elementi del contenitore in quanto vogliamo confrontare la distanza dalla media. Questo duplicazione però potrebbe portare a banali errori di copia e incolla, o di distrazione della logica del codice.

L'idea alla base di `std::ranges::sort`, la funzione di sorting resa disponibile dalla libreria `std::ranges`, permette di scorporare la logica di confronto degli elementi del vettore con la logica di trasformazione degli elementi del vettore stesso. Porta con sé inoltre un altro vantaggio, non indicare esplicitamente gli iteratori di inizio e fine del contenitore, ma passare direttamente il contenitore stesso.

```cpp
// ...

int main(int argc, char **argv) {
    // ...

    double sum = std::accumulate(vec.begin(), vec.end(), 0.0);
    double average = sum / (double)vec.size();
    std::ranges::sort(vec, {}, [&average](int32_t number) { return std::abs(number - average); });

    // ...
}
```

Il primo parametro di `std::ranges::sort` riguarda il contenitore da ordinare, il secondo parametro riguarda la funzione di confronto, la quale può essere omessa in quanto di default viene utilizzata `std::ranges::less`, mentre il terzo parametro riguarda la funzione di projection, la quale trasforma gli elementi del contenitore prima di passarli alla funzione di confronto. In questo modo, ogni elemento del vettore viene confrontato con gli altri elementi attraverso la "chiave" creata dalla funzione di proiezione, ovvero la distanza dalla media. Possiamo pensare a questa funzione come il corrispettivo di Python `sorted(list, key=function)`.

Il concetto di "chiave", "indice", o semplicemente valore di confronto può essere portato ancora più all'estremo con il codice di seguito.

```cpp
// ...

struct entry {
    int32_t value_;
    double distance_;
};

int main(int argc, char **argv) {
    // ...

    double sum = std::accumulate(vec.begin(), vec.end(), 0.0);
    double average = sum / (double)vec.size();

    std::vector<entry> entries;
    std::transform(vec.begin(), vec.end(), std::back_inserter(entries),
                    [&average](int32_t number) { return entry{number, std::abs(number - average)}; });

    std::ranges::sort(entries, {}, &entry::distance_);

    // ...
}
```

 Supponiamo di voler creare un secondo vettore composto da oggetti di tipo `entry`, semplicemente una classe contenente al suo interno il valore dell'elemento del vettore originale e la distanza dalla media. Supponiamo inoltre di voler ordinare questo vettore per la distanza dalla media e lasciare il vettore originale ordinato per valore crescente.

Per fare ciò creiamo il nostro vettore di entry utilizzando la funzione `std::transform` a partire dal vettore originale `vec`. La funzione `std::transform` svolge praticamente lo stesso compito di `std::copy`, ma permette di trasformare gli elementi del contenitore prima di copiarli nel nuovo contenitore. In questo caso, la funzione di trasformazione è una lambda expression che accetta un elemento del vettore originale e restituisce un oggetto di tipo `entry` con il valore dell'elemento e la distanza dalla media.

Ora possiamo applicare l'ordinamento sul nuovo vettore di entries utilizzando `std::ranges::sort`. La particolarità in questo caso è che la proiezione non è altro che un puntatore a membro della classe `entry`, ovvero `&entry::distance_`. In questo modo, la funzione di proiezione restituisce il valore della distanza dalla media per ogni elemento del vettore di entries, e la funzione di confronto utilizza questi valori per ordinare gli elementi del vettore.

Questo mostra molto bene come la proiezione possa essere utilizzata come una sorta di "chiave" per ordinare gli elementi di un contenitore, senza dover modificare il contenitore originale o creare funzioni di confronto complesse.

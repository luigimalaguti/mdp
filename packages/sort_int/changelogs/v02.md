### From code

```c
#include <assert.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>

int compare_int32(const void *pointer_a, const void *pointer_b) {
    int32_t number_a = *(int32_t *)pointer_a;
    int32_t number_b = *(int32_t *)pointer_b;
    return (number_a > number_b) - (number_a < number_b);
}

typedef struct vector {
    size_t size_;
    size_t capacity_;
    int32_t *data_;
} vector_t;

vector_t *vector_constructor(vector_t *this) {
    this->size_ = 0;
    this->capacity_ = 0;
    this->data_ = NULL;
    return this;
}

vector_t *vector_destructor(vector_t *this) {
    free(this->data_);
    return this;
}

vector_t *vector_new(void) {
    vector_t *this = malloc(sizeof(vector_t));
    return vector_constructor(this);
}

void vector_delete(vector_t *this) {
    free(vector_destructor(this));
}

void vector_push_back(vector_t *this, int32_t number) {
    if (this->size_ >= this->capacity_) {
        size_t temp_capacity = this->capacity_ == 0 ? 1 : this->capacity_ * 2;
        int32_t *temp_data = realloc(this->data_, temp_capacity * sizeof(int32_t));
        this->data_ = temp_data;
        this->capacity_ = temp_capacity;
    }
    this->data_[this->size_] = number;
    this->size_++;
}

void vector_sort(vector_t *this) {
    qsort(this->data_, this->size_, sizeof(int32_t), compare_int32);
}

size_t vector_size(const vector_t *this) {
    return this->size_;
}

int32_t vector_at(const vector_t *this, size_t index) {
    assert(index < this->size_);
    return this->data_[index];
}

int main(int argc, char **argv) {
    if (argc != 3) {
        printf("Usage: %s <input_file> <output_file>\n", argv[0]);
        return EXIT_FAILURE;
    }

    const char *input_filename = argv[1];
    const char *output_filename = argv[2];

    FILE *input_file = fopen(input_filename, "r");
    if (input_file == NULL) {
        printf("Error: Could not open file %s\n", input_filename);
        return EXIT_FAILURE;
    }

    vector_t *vec = vector_new();

    int32_t number;
    while (fscanf(input_file, "%d", &number) == 1) {
        vector_push_back(vec, number);
    }
    fclose(input_file);

    vector_sort(vec);

    FILE *output_file = fopen(output_filename, "w");
    if (output_file == NULL) {
        vector_delete(vec);
        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    for (size_t index = 0; index < vector_size(vec); index++) {
        fprintf(output_file, "%d\n", vector_at(vec, index));
    }
    fclose(output_file);

    vector_delete(vec);
    return EXIT_SUCCESS;
}
```

## Details

### Include standard library C

Quando in un programma in C++ si vuole utilizzare una libreria standard del C, è necessario includere l'header della libreria C, come avviene normalmente in C. Questa include può avvenire in due differenti modi. Il primo è il classico include utilizzando la sintassi C.

```c
#include <assert.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>

// ...
```

Il secondo modo utilizza la convenzione standard in C++ che prevede di anteporre la lettera `c` al nome della libreria e di omettere l'estensione `.h`.

```cpp
#include <cassert>
#include <cstdint>
#include <cstdio>
#include <cstdlib>

// ...
```

### Struttura del vettore

Nella transizione tra C e C++ la struttura del vettore subisce qualche cambiamento. Come prima cosa, in C++ non è più necessario definisce esplicitamente il tipo di dato attraverso `typedef`. Il compilatore C++ permette di utilizzare direttamente il nome della struttura come tipo di dato.

La seconda differenza riguarda la definizione dei metodi. In C++ i metodi vengono definiti all'interno della struttura, mentre in C vengono definiti al di fuori della struttura e richiedono un puntatore alla struttura come primo parametro. Il fatto di definire i metodi all'interno della `struct` non significa però che essi siano memorizzati all'interno della struttura stessa. In realtà, i metodi sono memorizzati in un'area separata della memoria e non occupano spazio all'interno della struttura. Infatti, la struttura `vector` avrà lo stesso spazio occupato dalla stessa struttura in C, ovvero di 24 byte, 8 per ciascun membro della struttura.

Dal momento in cui i metodi sono definiti all'interno della struttura, la nostra precedente convenzione di utilizzare il prefix `vector_` per i nomi dei metodi non è più necessaria. Infatti, il compilatore C++ è in grado di distinguere i metodi della struttura `vector` da eventuali funzioni esterne con lo stesso nome. Per questo motivo, possiamo rimuovere il prefisso `vector_` dai nomi dei metodi.

#### Keyword this

Se a questo punto proviamo a compilare il codice così com'è, il compilatore ci restituirà un errore.

```c
// ...

struct vector {
    size_t size_;
    size_t capacity_;
    int32_t *data_;

    vector_t *constructor(vector_t *this) {
        this->size_ = 0;
        this->capacity_ = 0;
        this->data_ = NULL;
        return this;
    }

    vector_t *destructor(vector_t *this) {
        free(this->data_);
        return this;
    }

    vector_t *new(void) {
        vector_t *this = malloc(sizeof(vector_t));
        return vector_constructor(this);
    }

    void delete(vector_t *this) {
        free(vector_destructor(this));
    }

    void push_back(vector_t *this, int32_t number) {
        if (this->size_ >= this->capacity_) {
            size_t temp_capacity = this->capacity_ == 0 ? 1 : this->capacity_ * 2;
            int32_t *temp_data = realloc(this->data_, temp_capacity * sizeof(int32_t));
            this->data_ = temp_data;
            this->capacity_ = temp_capacity;
        }
        this->data_[this->size_] = number;
        this->size_++;
    }

    void sort(vector_t *this) {
        qsort(this->data_, this->size_, sizeof(int32_t), compare_int32);
    }

    size_t size(const vector_t *this) {
        return this->size_;
    }

    int32_t at(const vector_t *this, size_t index) {
        assert(index < this->size_);
        return this->data_[index];
    }
};

// ...
```

Infatti in C++ la keyword `this` è riservata e non può essere utilizzata come nome di un parametro. Il motivo è semplice, definendo le funzioni all'interno della `struct` come metodi il C++ passa implicitamente come primo parametro `this`, ovvero un puntatore alla struttura stessa. In questo modo, non è più necessario passare esplicitamente il puntatore alla struttura come primo parametro della funzione. Per risolvere l'errore, è sufficiente rimuovere il parametro `this` dalle funzioni.

#### Metodi con this costante

Rimuovendo `this` dai parametri delle funzioni, noteremo come potremmo trovarci confusi sul come procedere in questo compito. Infatti, i metodi come `at` oppure `size`, definivano il parametro `this` come costante, in modo da non poter modificare i membri della struttura. In C++ questo concetto viene espresso attraverso la keyword `const` posta dopo la dichiarazione della funzione. Questo vuole indicare che il parametro passato implicitamente come primo parametro, ovvero `this`, è costante e non può essere modificato.

```c
// ...

struct vector {
    size_t size_;
    size_t capacity_;
    int32_t *data_;

    vector_t *constructor() {
        this->size_ = 0;
        this->capacity_ = 0;
        this->data_ = NULL;
        return this;
    }

    vector_t *destructor() {
        free(this->data_);
        return this;
    }

    vector_t *new() {
        vector_t *this = malloc(sizeof(vector_t));
        return vector_constructor(this);
    }

    void delete() {
        free(vector_destructor(this));
    }

    void push_back(int32_t number) {
        if (this->size_ >= this->capacity_) {
            size_t temp_capacity = this->capacity_ == 0 ? 1 : this->capacity_ * 2;
            int32_t *temp_data = realloc(this->data_, temp_capacity * sizeof(int32_t));
            this->data_ = temp_data;
            this->capacity_ = temp_capacity;
        }
        this->data_[this->size_] = number;
        this->size_++;
    }

    void sort() {
        qsort(this->data_, this->size_, sizeof(int32_t), compare_int32);
    }

    size_t size() const {
        return this->size_;
    }

    int32_t at(size_t index) const {
        assert(index < this->size_);
        return this->data_[index];
    }
};

// ...
```

Inoltre, l'utilizzo di `this` all'interno dei metodi per fare riferimento ai membri della struttura non è più necessario. Infatti, il compilatore C++ è in grado di distinguere i membri della struttura dai parametri delle funzioni e dalle variabili locali. Il compilatore C++ cerca prima tra i parametri della funzione e le variabili locali, e se non trova il nome cercato, cerca tra i membri della struttura, passando eventualmente allo scope esterno se non presente. Per questo motivo, possiamo rimuovere `this->` dai membri della struttura.

#### Costruttore e distruttore

In C++ quando si definisce una struttura, il compilatore genera automaticamente un costruttore e un distruttore di default. Il costruttore viene chiamato automaticamente quando si crea un'istanza della struttura, mentre il distruttore viene chiamato automaticamente quando l'istanza della struttura viene distrutta nel caso in cui venga allocata nello stack. Sebbene il costruttore e il distruttore di default siano generati automaticamente, è possibile definire un costruttore e un distruttore grazie al concetto di overloading.

Il costruttore e distruttore sono metodi particolari. Devono essere definiti con lo stesso nome della struttura e non devono avere un tipo di ritorno. Il costruttore può avere parametri, mentre il distruttore non può avere parametri. Il distruttore viene definito anteponendo il simbolo `~` al nome della struttura.

```cpp
// ...

struct vector {
    size_t size_;
    size_t capacity_;
    int32_t *data_;

    vector() {
        size_ = 0;
        capacity_ = 0;
        data_ = NULL;
    }

    ~vector() {
        free(data_);
    }

    // ...
};

// ...
```

Mentre i metodi `new` e `delete` sono stati rimossi, in quanto non più necessari. Infatti, in C++ è possibile creare un'istanza della struttura direttamente nello stack, senza dover allocare memoria dinamicamente. In questo caso, il costruttore viene chiamato automaticamente quando l'istanza della struttura viene creata, mentre il distruttore viene chiamato automaticamente quando l'istanza della struttura viene distrutta. Anche nel caso in cui si voglia allocare dinamicamente la memoria per un'istanza della struttura, è possibile utilizzare l'operatore `new` e `delete` di C++, che chiamano automaticamente il costruttore e il distruttore.

#### Casting esplicito

Una differenza tra C e C++ riguarda il casting. In C++ il casting è più restrittivo rispetto a C. Infatti, in C++ non è possibile convertire un puntatore `void *` in un puntatore di un tipo specifico senza utilizzare un cast esplicito. Per questo motivo, il compilatore solleverà degli errori in tutti i `realloc` che abbiamo definito. Per risolvere questo problema, è necessario utilizzare un cast esplicito per convertire il puntatore `void *` restituito da `realloc` in un puntatore di tipo `int32_t *`.

```c
// ...

struct vector {
    // ...

    void push_back(int32_t number) {
        if (this->size_ >= this->capacity_) {
            size_t temp_capacity = this->capacity_ == 0 ? 1 : this->capacity_ * 2;
            int32_t *temp_data = (int32_t *)realloc(this->data_, temp_capacity * sizeof(int32_t));
            this->data_ = temp_data;
            this->capacity_ = temp_capacity;
        }
        this->data_[this->size_] = number;
        this->size_++;
    }

    // ...
};

// ...
```

### Implementazione del main

A questo punto dobbiamo modificare il `main` per utilizzare la nuova struttura del vettore. In C++ possiamo creare un'istanza della struttura direttamente nello stack, senza dover allocare memoria dinamicamente. In questo caso, il costruttore viene chiamato automaticamente quando l'istanza della struttura viene creata, mentre il distruttore viene chiamato automaticamente quando l'istanza della struttura viene distrutta.

```cpp
// ...

int main(int argc, char **argv) {
    // ...

    vector vec;

    int32_t number;
    while (fscanf(input_file, "%d", &number) == 1) {
        vec.push_back(number);
    }
    fclose(input_file);

    vec.sort();

    FILE *output_file = fopen(output_filename, "w");
    if (output_file == NULL) {
        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    for (size_t index = 0; index < vec.size(); index++) {
        fprintf(output_file, "%d\n", vec.at(index));
    }
    fclose(output_file);

    return EXIT_SUCCESS;
}
```

Ma possiamo anche allocare dinamicamente la memoria per un'istanza della struttura, utilizzando l'operatore `new`. In questo caso, il costruttore viene chiamato automaticamente quando l'istanza della struttura viene creata, mentre il distruttore viene chiamato automaticamente quando l'istanza della struttura viene distrutta. Per deallocare la memoria, è necessario utilizzare l'operatore `delete`, che chiama automaticamente il distruttore.

```cpp
// ...

int main(int argc, char **argv) {
    // ...

    vector *vec = new vector();

    int32_t number;
    while (fscanf(input_file, "%d", &number) == 1) {
        vec->push_back(number);
    }
    fclose(input_file);

    vec->sort();

    FILE *output_file = fopen(output_filename, "w");
    if (output_file == NULL) {
        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    for (size_t index = 0; index < vec->size(); index++) {
        fprintf(output_file, "%d\n", vec->at(index));
    }
    fclose(output_file);

    delete vec;
    return EXIT_SUCCESS;
}
```

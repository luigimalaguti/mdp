```cpp
#include <cassert>
#include <cstdint>
#include <cstdio>
#include <cstdlib>
#include <utility>

namespace mdp {
    int compare_int32(const void *pointer_a, const void *pointer_b) {
        int32_t number_a = *(int32_t *)pointer_a;
        int32_t number_b = *(int32_t *)pointer_b;
        return (number_a > number_b) - (number_a < number_b);
    }

    class vector {
    private:
        size_t size_;
        size_t capacity_;
        int32_t *data_;

    public:
        vector() {
            size_ = 0;
            capacity_ = 0;
            data_ = nullptr;
        }

        vector(size_t capacity) {
            size_ = 0;
            capacity_ = capacity;
            data_ = new int32_t[capacity];
        }

        vector(const vector &other) {
            size_ = other.size_;
            capacity_ = other.capacity_;
            data_ = new int32_t[capacity_];
            for (size_t index = 0; index < size_; index++) {
                data_[index] = other.data_[index];
            }
        }

        vector(vector &&other) {
            size_ = other.size_;
            capacity_ = other.capacity_;
            data_ = other.data_;
            other.data_ = nullptr;
        }

        ~vector() {
            delete[] data_;
        }

        vector &operator=(vector rhs) {
            swap(*this, rhs);
            return *this;
        }

        void push_back(int32_t number) {
            if (size_ >= capacity_) {
                size_t temp_capacity = capacity_ == 0 ? 1 : capacity_ * 2;
                int32_t *temp_data = new int32_t[temp_capacity];
                for (size_t index = 0; index < size_; index++) {
                    temp_data[index] = data_[index];
                }
                delete[] data_;
                data_ = temp_data;
                capacity_ = temp_capacity;
            }
            data_[size_] = number;
            size_++;
        }

        void sort() {
            qsort(data_, size_, sizeof(int32_t), compare_int32);
        }

        size_t size() const {
            return size_;
        }

        int32_t at(size_t index) const {
            assert(index < size_);
            return data_[index];
        }

        bool empty() const {
            return size_ == 0;
        }

        friend void swap(vector &vector_a, vector &vector_b) {
            using std::swap;
            swap(vector_a.size_, vector_b.size_);
            swap(vector_a.capacity_, vector_b.capacity_);
            swap(vector_a.data_, vector_b.data_);
        }
    };
}  // namespace mdp

mdp::vector read_from_file(const char *filename, size_t capacity) {
    mdp::vector vec(capacity);
    FILE *file = fopen(filename, "r");
    if (file != NULL) {
        int32_t number;
        while (fscanf(file, "%d", &number) == 1) {
            vec.push_back(number);
        }
        fclose(file);
    }
    return vec;
}

bool write_to_file(const char *filename, const mdp::vector &vec) {
    FILE *file = fopen(filename, "w");
    if (file == NULL) {
        return false;
    }
    for (size_t index = 0; index < vec.size(); index++) {
        fprintf(file, "%d\n", vec.at(index));
    }
    fclose(file);
    return true;
}

int main(int argc, char **argv) {
    if (argc != 3) {
        printf("Usage: %s <input_file> <output_file>\n", argv[0]);
        return EXIT_FAILURE;
    }

    const char *input_filename = argv[1];
    const char *output_filename = argv[2];

    mdp::vector vec;
    vec = read_from_file(input_filename, 10);
    if (vec.empty() == true) {
        printf("Error: Could not open file %s\n", input_filename);
        return EXIT_FAILURE;
    }

    mdp::vector sorted = vec;
    vec.sort();
    swap(vec, sorted);

    bool result = write_to_file(output_filename, sorted);

        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

## Details

### Templated class

A questo punto il codice risulta a un buon punto. Ma abbiamo un grosso limite. La nostra classe `vector` può essere utilizzata solamente con variabili di tipo `int32_t`. Non possiamo usare `double`, `float`, `char` o qualsiasi altro tipo di dato. Per risolvere questo problema, possiamo trasformare la nostra classe in una **templated class**.

Una classe templated è una classe che può essere utilizzata con qualsiasi tipo di dato. Per fare questo, dobbiamo aggiungere un parametro di template alla nostra classe. In questo caso, possiamo chiamare il parametro `T`.

```cpp
namespace mdp {
    // ...

    template <typename T>
    class vector {
        // ...
    };
}  // namespace mdp

// ...
```

Con questa sintassi `template <typename>` stiamo definendo una classe generica che può essere utilizzata con qualsiasi tipo di dato. Il parametro `T` rappresenta il tipo di dato che verrà utilizzato nella classe. Ora che abbiamo definito un tipo generico per la classe, possiamo andare ad applicare questo tipo alle variabili della classe che lo richiedono.

```cpp
namespace mdp {
    // ...

    template <typename T>
    class vector {
    private:
        size_t size_;
        size_t capacity_;
        T *data_;

    public:
        // ...

        vector(size_t capacity) {
            size_ = 0;
            capacity_ = capacity;
            data_ = new T[capacity];
        }

        vector(const vector &other) {
            size_ = other.size_;
            capacity_ = other.capacity_;
            data_ = new T[capacity_];
            for (size_t index = 0; index < size_; index++) {
                data_[index] = other.data_[index];
            }
        }

        // ...

        void push_back(const T &number) {
            if (size_ >= capacity_) {
                size_t temp_capacity = capacity_ == 0 ? 1 : capacity_ * 2;

                    temp_data[index] = data_[index];
                }
                delete[] data_;
                data_ = temp_data;
                capacity_ = temp_capacity;
            }
            data_[size_] = number;
            size_++;
        }

        // ...

        const T &at(size_t index) const {
            assert(index < size_);
            return data_[index];
        }

        // ...
    };
}  // namespace mdp

mdp::vector<int32_t> read_from_file(const char *filename, size_t capacity) {
    mdp::vector<int32_t> vec(capacity);

    // ...
}

bool write_to_file(const char *filename, const mdp::vector<int32_t> &vec) {
    // ...
}

int main(int argc, char **argv) {

    mdp::vector<int32_t> vec;
```

Sembrano tante modifiche, ma in realtà le modifiche sono solamente due. La prima, e quella più intuitiva, è stata quella di modificare il tipo di dato di `data_` della classe per fare in modo di non essere più solamente `int32_t`, ma di essere definito come il tipo generico che abbiamo appena definito `T`. Allo stesso modo abbiamo modificato tutti i parametri dei metodi e relativi valori di ritorno che lo richiedevano.

La seconda modifica, invece, riguarda non più la classe `vector`, ma le funzioni di supporto `read_from_file`, `write_to_file` e il codice del `main`. Infatti, è vero che ora la classe `vector` può supportare un tipo generico di dato, ma quando si deve utilizzare questa classe è necessario indicare con quali tipi di dato la classe deve lavorare. E per fare ciò si definisce il tipo di dato come abbiamo sempre fatto `mdp::vector` seguito però da `<int32_t>`. Questo indica che la classe `vector` deve essere utilizzata con il tipo di dato `int32_t`.

Il meccanismo con cui il compilatore gestisce le classi templated è molto potente, ma anche molto complesso. Infatti, il compilatore genera una nuova classe per ogni tipo di dato che viene utilizzato con la classe templated. Questo significa che se utilizziamo la classe `vector` con `int32_t`, il compilatore genererà una nuova classe `vector<int32_t>`. Se utilizziamo la classe `vector` con `double`, il compilatore genererà una nuova classe `vector<double>`. E così via.

Possiamo notare, inoltre, come abbiamo modificato leggermente anche i metodi `push_back` e `at` impostando rispettivamente il parametro `number` e valore di ritorno come riferimento costante. Questo perché, in questo modo, possiamo evitare di copiare il valore passato come parametro o il valore ritornato dal metodo. Infatti, finché parliamo di tipi di dato primitivi come `int32_t`, `double`, `float`, `char` e così via, la copia del valore non comporta un grande overhead. Ma se parliamo di tipi di dato complessi come classi o strutture, la copia del valore può comportare un grande overhead. Per questo motivo, è buona norma passare i parametri per riferimento costante e ritornare i valori per riferimento costante.

#### Templated function

I più attenti potrebbero aver notato che durante il nostro processo di modifica della classe `vector` per renderla templated abbiamo omesso un metodo in particolare. Questo metodo è `sort`.

Il metodo `sort` merita un pò di attenzione in più nel momento in cui vogliamo renderlo templated. Infatti, il metodo `sort` utilizza la funzione `qsort` della libreria standard del C. Questa funzione richiede un puntatore a funzione che confronti due elementi del tipo di dato che vogliamo ordinare. In questo caso, la funzione `compare_int32` confronta due elementi di tipo `int32_t`. Ma se vogliamo rendere il metodo `sort` templated, dobbiamo rendere anche la funzione di confronto templated. Questa modifica avviene allo stesso modo di quanto fatto per la classe `vector`.

```cpp
namespace mdp {
    template <typename T>
    int compare(const void *pointer_a, const void *pointer_b) {
        T value_a = *(T *)pointer_a;
        T value_b = *(T *)pointer_b;
        return (value_a > value_b) - (value_a < value_b);
    }

    template <typename T>
    class vector {
        // ...

        void sort() {
            qsort(data_, size_, sizeof(T), compare<T>);
        }

        // ...
    };
}  // namespace mdp

// ...
```

Abbiamo definito un nuovo metodo `compare` templated che confronta due elementi del tipo `T`. Questo metodo viene utilizzato dal metodo `sort` per ordinare gli elementi del vettore. Quando specifichiamo questa funzione come parametro di `qsort`, dobbiamo indicarne anche il tipo di dato con cui la funzione deve lavorare, che in questo caso sarà lo stesso tipo di dato con cui la classe `vector` è stata istanziata.

Questo meccanismo di sorting, però, solleva una grossa domanda. Come possiamo fare a ordinare oggetti di qualsiasi tipologia di dato? Ad esempio una `struct` il compilatore non sa come deve essere ordinata, non sa come utilizzare la funzione `compare` ovvero non sa come applicare `>` e `<` per tipologie di dati complesse.

Per risolvere questo problema, le tipologie di dati che si vogliono utilizzare **devono** avere definito appunto gli operatori di confronto `>` e `<`. In questo modo, il compilatore sa come confrontare due oggetti di quel tipo di dato e può utilizzare la funzione `compare` per ordinare gli elementi del vettore. Se questi operatori non sono definiti allora verrà generato un errore di compilazione. Ovviamente se si utilizzano tipi di dato primitivi questi operatori sono già definiti e non ci sarà alcun problema.

Un aspetto interessante è che seppur questi operatori non vengano definiti, e quindi non avere la possibilità di ordinare gli elementi del vettore, il compilatore non genera nessun errore se questo sorting non viene eseguito. Questo ci fa capire come il compilatore generi il codice della classe templated solo quando viene effettivamente utilizzata. In questo caso, se non viene mai chiamato il metodo `sort`, il compilatore non genererà il codice per la funzione `compare` e quindi non ci sarà alcun problema.

### Ottimizzazioni

Il codice è completo e funziona correttamente. Possiamo però migliorarlo su tre fronti, il modo in cui i costruttori inizializzano gli attributi, la duplicazione tra costruttore di default e costruttore con capacità, e l'assenza di un modo naturale per accedere e modificare gli elementi del vettore.

#### Constructor initialization lists

Per capire perché questo argomento è importante, conviene partire da un esempio più semplice del nostro `vector`. Immaginiamo una classe che avvolge un `int`, memorizzandolo però per riferimento.

```cpp
class int_wrapper {
    int &value_;

public:
    int_wrapper() {
        value_ = 0;
    }

    void print() {
        printf("%d\n", value_);
    }
};

int main(void) {
    int_wrapper wrap;

    return 0;
}
```

Compilando questo codice otteniamo un errore `error: uninitialized reference member in 'int&'`. Il motivo è che `value_` è un riferimento `int &`, e in C++ un riferimento **deve** essere legato a qualcosa nel momento stesso in cui viene creato, non può esistere "vuoto" nemmeno per un istante, e una volta legato non può più essere fatto puntare altrove. La riga `value_ = 0` dentro il corpo del costruttore non è un'inizializzazione, è un'assegnazione, esattamente come avevamo distinto in una versione precedente del codice. Ma un'assegnazione presuppone che `value_` sia già stato inizializzato, cosa che in questo caso non è mai avvenuta.

Proviamo a passare il valore come parametro del costruttore, sperando che questo risolva il problema.

```cpp
class int_wrapper {
    int &value_;

public:
    int_wrapper(int &value) {
        value_ = value;
    }

    void print() {
        printf("%d\n", value_);
    }
};

int main(void) {
    int value = 123;
    int_wrapper wrap(value);

    return 0;
}
```

L'errore persiste identico, `value_ = value` è ancora un'assegnazione nel corpo del costruttore, non un'inizializzazione. Il problema non è cosa stiamo assegnando, ma quando. Il corpo del costruttore viene eseguito solo **dopo** che tutti gli attributi sono già stati creati, e per un riferimento questo significa che dev'essere già stato legato a qualcosa prima ancora di arrivare al corpo.

La soluzione è la **constructor initialization list**, una sintassi che permette di inizializzare gli attributi nel momento esatto in cui vengono creati, prima che il corpo del costruttore venga eseguito.

```cpp
class int_wrapper {
    int &value_;

public:
    int_wrapper(int &value) : value_(value) {}

    void print() {
        printf("%d\n", value_);
    }
};

int main(void) {
    int value = 123;
    int_wrapper wrap(value);
    wrap.print(); // 123

    return 0;
}
```

Con `: value_(value)` tra la firma del costruttore e il corpo, `value_` viene legato a `value` durante l'inizializzazione, ed il codice compila ed esegue correttamente.

Applichiamo la stessa sintassi ai costruttori di `vector`:

```cpp
namespace mdp {
    // ...

    template <typename T>
    class vector {
    private:
        size_t size_;
        size_t capacity_;
        T *data_;

    public:
        vector() : size_(0), capacity_(0), data_(nullptr) {}

        vector(size_t capacity) : size_(0), capacity_(capacity), data_(new T[capacity]) {}

        vector(const vector &other) : size_(other.size_), capacity_(other.capacity_), data_(new T[capacity_]) {
            for (size_t index = 0; index < size_; index++) {
                data_[index] = other.data_[index];
            }
        }

        vector(vector &&other) : size_(other.size_), capacity_(other.capacity_), data_(other.data_) {
            other.data_ = nullptr;
        }

        // ...
    };
}  // namespace mdp

// ...
```

Per `vector`, `size_` e `capacity_` non sono riferimenti, quindi la constructor initialization list non è strettamente necessaria per far compilare il codice, è comunque preferibile, perché è lo stile idiomatico del C++ ed evita l'inutile doppio passaggio "crea con un valore, poi sovrascrivilo" che avviene assegnando nel corpo. Ma è soprattutto un'abitudine da avere pronta, se un domani un attributo della classe fosse un riferimento, una `const`, o un oggetto di un tipo senza costruttore di default, l'assegnazione nel corpo smetterebbe semplicemente di compilare, esattamente come nell'esempio di `int_wrapper`.

Un'ultima cosa da sapere sulla constructor initialization list, gli attributi vengono sempre inizializzati nell'**ordine in cui sono dichiarati nella classe**, non nell'ordine in cui compaiono nella lista. Per `vector`, questo significa che `size_` viene sempre inizializzato per primo, poi `capacity_`, infine `data_`, indipendentemente da come li scriviamo dopo i due punti.

#### Default parameter values

Guardando i due costruttori appena scritti, notiamo che sono quasi identici. `vector()` è semplicemente `vector(size_t capacity)` chiamato con `capacity = 0`. Il C++ permette di evitare questa duplicazione assegnando un **valore di default** a un parametro. Se chi chiama la funzione non specifica quel parametro, viene usato automaticamente il valore di default.

```cpp
namespace mdp {
    // ...

    template <typename T>
    class vector {
    private:
        size_t size_;
        size_t capacity_;
        T *data_;

    public:
        vector(size_t capacity = 0) : size_(0), capacity_(capacity), data_(nullptr) {
            if (capacity_ > 0) {
                data_ = new T[capacity_];
            }
        }

        // ...
    };
}  // namespace mdp

// ...
```

Un solo costruttore ora copre entrambi i casi, `vector<int> a` usa `capacity = 0`, il default, mentre `vector<int> b(10)` usa `capacity = 10`. Il vecchio costruttore di default va rimosso, avendo la stessa identica firma di `vector(size_t capacity = 0)` una volta chiamato senza argomenti, il compilatore non saprebbe più quale dei due usare.

Da notare anche come, in questo costruttore, inizializzazione e assegnazione convivano fianco a fianco. `data_` viene inizializzato a `nullptr` nella constructor initialization list, e solo successivamente, nel corpo, viene eventualmente sovrascritto con un array vero se `capacity_` è maggiore di zero. Questo è corretto, l'inizializzazione a `nullptr` è economica, non alloca nulla, quindi non c'è nessun problema nel farla comunque e poi eventualmente sostituirla.

#### Access operators

Un'ultima lacuna del nostro `vector`, non possiamo accedere ai suoi elementi con la sintassi `[]`, quella familiare degli array e, appunto, dei vettori. Come per ogni altro operatore già visto, la soluzione è definirne l'overload.

```cpp
namespace mdp {
    // ...

    template <typename T>
    class vector {
        // ...

        const T &operator[](size_t index) const {
            return data_[index];
        }

        // ...
    };
}  // namespace mdp

// ...
```

Questa versione richiede che `this` sia `const`, l'operatore non modifica il vettore, e restituisce un riferimento costante `const T &`, per evitare di copiare `T` che potrebbe essere un tipo costoso da copiare, senza permettere che venga modificato dall'esterno. È, di fatto, una versione di `at()` che salta il controllo dei limiti `assert(index < size_)`, `at()` verifica che l'indice sia valido, `operator[]` no, esattamente come accade in `std::vector`.

Con questo solo operatore, però, non possiamo ancora scrivere `vec[0] = 1`, proviamo e il compilatore rifiuta l'assegnazione, perché il valore restituito è `const`. La soluzione non è modificare l'operatore esistente, ma aggiungerne un secondo, con una firma diversa quanto basta perché il compilatore possa distinguerli:

```cpp
namespace mdp {
    // ...

    template <typename T>
    class vector {
        // ...

        const T &operator[](size_t index) const {
            return data_[index];
        }

        T &operator[](size_t index) {
            return data_[index];
        }

        // ...
    };
}  // namespace mdp

// ...
```

La differenza tra i due sta solo nella costanza, togliendo `const` da `this` e dal valore di ritorno, otteniamo una versione dell'operatore pensata per la scrittura, che restituisce un riferimento modificabile. Il compilatore sceglie automaticamente la versione giusta in base al contesto, se il vettore su cui viene chiamato `[]` è `const`, viene usata la prima versione, sola lettura, altrimenti viene usata la seconda, che permette anche la scrittura.

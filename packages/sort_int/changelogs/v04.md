```cpp
#include <cassert>
#include <cstdint>
#include <cstdio>
#include <cstdlib>

int compare_int32(const void *pointer_a, const void *pointer_b) {
    int32_t number_a = *(int32_t *)pointer_a;
    int32_t number_b = *(int32_t *)pointer_b;
    return (number_a > number_b) - (number_a < number_b);
}

struct vector {
    size_t size_;
    size_t capacity_;
    int32_t *data_;

    vector() {
        size_ = 0;
        capacity_ = 0;
        data_ = NULL;
    }

    vector(size_t capacity) {
        size_ = 0;
        capacity_ = capacity;
        data_ = new int32_t[capacity];
    }

    vector(const vector &other) {
        size_ = other.size_;
        capacity_ = other.capacity_;
        data_ = new int32_t[capacity_];
        for (size_t index = 0; index < size_; index++) {
            data_[index] = other.data_[index];
        }
    }

    ~vector() {
        delete[] data_;
    }

    vector &operator=(const vector &rhs) {
        if (this != &rhs) {
            size_ = rhs.size_;
            capacity_ = rhs.capacity_;
            delete[] data_;
            data_ = new int32_t[capacity_];
            for (size_t index = 0; index < size_; index++) {
                data_[index] = rhs.data_[index];
            }
        }
        return *this;
    }

    void push_back(int32_t number) {
        if (size_ >= capacity_) {
            size_t temp_capacity = capacity_ == 0 ? 1 : capacity_ * 2;
            int32_t *temp_data = new int32_t[temp_capacity];
            for (size_t index = 0; index < size_; index++) {
                temp_data[index] = data_[index];
            }
            delete[] data_;
            data_ = temp_data;
            capacity_ = temp_capacity;
        }
        data_[size_] = number;
        size_++;
    }

    void sort() {
        qsort(data_, size_, sizeof(int32_t), compare_int32);
    }

    size_t size() const {
        return size_;
    }

    int32_t at(size_t index) const {
        assert(index < size_);
        return data_[index];
    }

    bool empty() const {
        return size_ == 0;
    }
};

vector read_from_file(const char *filename, size_t capacity) {
    vector vec(capacity);
    FILE *file = fopen(filename, "r");
    if (file != NULL) {
        int32_t number;
        while (fscanf(file, "%d", &number) == 1) {
            vec.push_back(number);
        }
        fclose(file);
    }
    return vec;
}

bool write_to_file(const char *filename, const vector &vec) {
    FILE *file = fopen(filename, "w");
    if (file == NULL) {
        return false;
    }
    for (size_t index = 0; index < vec.size(); index++) {
        fprintf(file, "%d\n", vec.at(index));
    }
    fclose(file);
    return true;
}

int main(int argc, char **argv) {
    if (argc != 3) {
        printf("Usage: %s <input_file> <output_file>\n", argv[0]);
        return EXIT_FAILURE;
    }

    const char *input_filename = argv[1];
    const char *output_filename = argv[2];

    vector vec;
    vec = read_from_file(input_filename, 10);
    if (vec.empty() == true) {
        printf("Error: Could not open file %s\n", input_filename);
        return EXIT_FAILURE;
    }

    vec.sort();

    bool result = write_to_file(output_filename, vec);
    if (result == false) {
        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

## Details

### Move constructor

Nell'ultima versione di codice prodotto abbiamo introdotto il copy constructor e l'assignment operator al fine di gestire correttamente la copia degli oggetti della classe `vector`. Infatti, senza di essi venivano utilizzati i metodi generati di default dal compilatore, i quali utilizzavano la shallow copy, sollevando di conseguenza errori di double free.

```cpp
// ...

int main(int argc, char **argv) {
    if (argc != 3) {
        printf("Usage: %s <input_file> <output_file>\n", argv[0]);
        return EXIT_FAILURE;
    }

    const char *input_filename = argv[1];
    const char *output_filename = argv[2];

    vector vec;
    vec = read_from_file(input_filename, 10);
    if (vec.empty() == true) {
        printf("Error: Could not open file %s\n", input_filename);
        return EXIT_FAILURE;
    }

    vec.sort();

    bool result = write_to_file(output_filename, vec);
    if (result == false) {
        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    return EXIT_SUCCESS;
}
```

Il codice nel `main` con ci eravamo lasciati si presentava come segue. Questo codice funziona correttamente, non abbiamo errori di double free e gli output attesi sono corretti. Ma questo codice ha un solo piccolo problema, l'eccesso di copie che vengono effettuate, come mostrato dal trace del debugger.

La prima copia avviene alla creazione del vettore definito all'interno del `main`, il secondo viene definito all'interno della funzione `read_from_file` indicandone una capacità iniziale. La terza copia avviene per via del copy constructor al fine di restituire il vettore interno a `read_from_file` per assegnarlo a `vec` interno al `main`. Mentre l'ultima copia avviene per mano dell'assignment operator, il quale duplica nuovamente i dati del vettore temporaneo restituito da `read_from_file` per assegnarli a `vec`. Questa pratica non è affatto ottimale, soprattutto se consideriamo il fine con cui vengono svolte, ovvero per assegnare un vettore completo di numeri letti da un file.

Per contenere tutto questo eccesso di copie, quello che hanno iniziato a fare tutti gli sviluppatori a partire da C++17 è stato quello di non separare più la definizione e l'assegnazione di un oggetto, ma di farlo in un'unica operazione, quindi di dichiarare e inizializzare allo stesso tempo un oggetto.

In questo modo viene utilizzata l'ottimizzazione entrata obbligatoriamente nello standard da C++, ovvero la **Guaranteed Copy Elision** che abbiamo introdotto e visto nella versione precedente. Infatti, in questa versione del codice non compiliamo più secondo lo standard C++11, ma C++23, e quindi il compilatore è obbligato ad applicare questa ottimizzazione.

In questo contesto nessuna delle due ottimizzazioni può essere applicata, e quindi il compilatore è costretto a fare una copia intermedia del vettore restituito da `read_from_file` per assegnarlo a `vec` come visto all'inizio.


A questo punto, per cercare di evitare un eccessivo numero di copie, idealmente dovremmo trovare il modo di "rubare" il contenuto del vettore temporaneo restituito da `read_from_file` per assegnarlo a `vec`, senza doverlo copiare.

In C++ questo è possibile grazie al **move constructor**. Il move constructor si differenzia dal copy constructor per la sua sintassi. Infatti, come parametro il move constructor non riceve solamente un riferimento a un oggetto, ma bensì un riferimento a un oggetto temporaneo, il quale si è certi che non verrà più utilizzato. Questo è possibile grazie al fatto che il parametro del move constructor è un **r-value reference**, ovvero un riferimento a un oggetto temporaneo, il quale non ha un nome e quindi non può essere riutilizzato.

```cpp
// ...

struct vector {
    // ...

    vector(vector &&other) {
        size_ = other.size_;
        capacity_ = other.capacity_;
        data_ = other.data_;
        other.data_ = nullptr;
    }

    //...
}

// ...
```

Il corpo del move constructor risulta essere già abbastanza esplicativo da solo. Dato che l'obbiettivo è quello di non copiare tutti i dati, ma di approfittare del fatto che l'oggetto venga distrutto in breve termine per "rubare", entrare in possesso, dei dati di esso per assegnarli al vettore corrente. Per svolgere questo processo, però, è essenziale fare in modo che il puntatore a `data_` dell'oggetto morente non punti più al nuovo oggetto che si sta costruendo, altrimenti si torna all'errore di double free. Per evitare ciò, quindi, si imposta il puntatore `data_` di `other` a `nullptr`, l'equivalente di `NULL` in C++. Questo passaggio fa sì che il parametro `vector &&other` non debba essere definito `const` come abbiamo fatto nel copy constructor, altrimenti non saremmo in grado di modificare il puntatore `data_` di `other`.

### Move assignment operator

Siamo riusciti a evitare una copia nel nostro codice. Abbiamo essenzialmente risolto il problema che risolverebbe la NRVO in automatico, ma attraverso l'implementazione del move constructor.

A questo punto potremmo chiederci se esiste un meccanismo analogo a quanto visto per il move constructor con il fine di ottenere il possesso del contenuto di un oggetto che sta per essere distrutto anche per l'oggetto temporaneo verso la variabile `vec` del `main`. E la risposta è sì, esiste.

L'idea è molto semplice. Se è possibile definire un costruttore che sia in grado di essere utilizzato in presenza di oggetti temporanei, allora potrebbe esistere anche un metodo di assegnamento che faccia lo stesso. Basterebbe modificare la signature dell'assignment operator già in nostro possesso per avere come parametro un r-value reference.

```cpp
// ...

struct vector {
    // ...

    vector &operator=(vector &&rhs) {
        size_ = rhs.size_;
        capacity_ = rhs.capacity_;
        delete[] data_;
        data_ = rhs.data_;
        rhs.data_ = nullptr;
        return *this;
    }

    // ...
}
```

Infatti, il **move assignment operator** è pressoché identico al move constructor, sia per la signature che per il codice, con due sole differenze.

La signature del move assignment operator deve restituire un riferimento all'oggetto corrente, in quanto dobbiamo avere la possibilità di concatenare più assegnamenti, come avviene per l'assignment operator. Inoltre, come abbiamo già visto, il parametro `vector &&rhs` deve essere un r-value reference non costante, al fine di poter modificare il puntatore di `data_` e non avere un double free.

La seconda differenza riguarda il corpo del metodo. In aggiunta al codice visto per il move constructor, in questo caso dobbiamo anche deallocare la memoria del left hand side `data_`. Se questo non lo facessimo, verrebbe modificato il puntatore `data_` per fare riferimento alla zona di memoria di `rhs.data_`, e quindi avremmo un **memory leak** su `data_` del left hand side. Deallocando `data_` prima della modifica, preveniamo appunto il memory leak.

In questo modo, definendo il **move constructor** e il **move assignment operator**, siamo stati in grado di evitare ben due copie degli oggetti, le quali coinvolgevano entrambe il vettore temporaneo. Abbiamo ottenuto attraverso questi due nuovi metodi il risultato che avremmo avuto con la NRVO attiva e la Guaranteed Copy Elision dell'inizializzazione. Ovviamente queste due ottimizzazioni attive risultano molto più efficaci, soprattutto in termini di operazioni svolte, ma siamo riusciti a evitare copie non necessarie.

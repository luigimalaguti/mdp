## Preview

### From code

```c
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>

int compare_int32(const void *pointer_a, const void *pointer_b) {
    int32_t number_a = *(int32_t *)pointer_a;
    int32_t number_b = *(int32_t *)pointer_b;
    return (number_a > number_b) - (number_a < number_b);
}

int main(int argc, char **argv) {
    if (argc != 3) {
        printf("Usage: %s <input_file> <output_file>\n", argv[0]);
        return EXIT_FAILURE;
    }

    const char *input_filename = argv[1];
    const char *output_filename = argv[2];

    FILE *input_file = fopen(input_filename, "r");
    if (input_file == NULL) {
        printf("Error: Could not open file %s\n", input_filename);
        return EXIT_FAILURE;
    }

    size_t size = 0;
    size_t capacity = 0;
    int32_t *data = NULL;

    int32_t number;
    while (fscanf(input_file, "%d", &number) == 1) {
        if (size >= capacity) {
            size_t temp_capacity = capacity == 0 ? 1 : capacity * 2;
            int32_t *temp_data = realloc(data, temp_capacity * sizeof(int32_t));
            data = temp_data;
            capacity = temp_capacity;
        }
        data[size] = number;
        size++;
    }
    fclose(input_file);

    qsort(data, size, sizeof(int32_t), compare_int32);

    FILE *output_file = fopen(output_filename, "w");
    if (output_file == NULL) {
        free(data);
        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    for (size_t index = 0; index < size; index++) {
        fprintf(output_file, "%d\n", data[index]);
    }
    fclose(output_file);

    free(data);
    return EXIT_SUCCESS;
}
```

### To code

```c
#include <assert.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>

int compare_int32(const void *pointer_a, const void *pointer_b) {
    int32_t number_a = *(int32_t *)pointer_a;
    int32_t number_b = *(int32_t *)pointer_b;
    return (number_a > number_b) - (number_a < number_b);
}

typedef struct vector {
    size_t size_;
    size_t capacity_;
    int32_t *data_;
} vector_t;

vector_t *vector_constructor(vector_t *this) {
    this->size_ = 0;
    this->capacity_ = 0;
    this->data_ = NULL;
    return this;
}

vector_t *vector_destructor(vector_t *this) {
    free(this->data_);
    return this;
}

vector_t *vector_new(void) {
    vector_t *this = malloc(sizeof(vector_t));
    return vector_constructor(this);
}

void vector_delete(vector_t *this) {
    free(vector_destructor(this));
}

void vector_push_back(vector_t *this, int32_t number) {
    if (this->size_ >= this->capacity_) {
        size_t temp_capacity = this->capacity_ == 0 ? 1 : this->capacity_ * 2;
        int32_t *temp_data = realloc(this->data_, temp_capacity * sizeof(int32_t));
        this->data_ = temp_data;
        this->capacity_ = temp_capacity;
    }
    this->data_[this->size_] = number;
    this->size_++;
}

void vector_sort(vector_t *this) {
    qsort(this->data_, this->size_, sizeof(int32_t), compare_int32);
}

size_t vector_size(const vector_t *this) {
    return this->size_;
}

int32_t vector_at(const vector_t *this, size_t index) {
    assert(index < this->size_);
    return this->data_[index];
}
```

## Details

### Struct vector

La transizione da un programma C standard verso un approccio orientato agli oggetti passa attraverso l'utilizzo delle `struct`. Essendo ancora il linguaggio C, l'approccio a oggetti mantiene ugualmente i classici costrutti C di base, ma questo permetterà di avere una transizione all'OOP molto più fluida.

Dovremo quindi creare una struttura che rappresenti la nostra entità, ovvero un vettore. In C è molto comune utilizzare `typedef` con le `struct` per creare un nuovo tipo di dato.

```c
// ...

typedef struct vector {
    size_t size_;
    size_t capacity_;
    int32_t *data_;
} vector_t;

// ...
```

La nostra struttura conterrà solamente le variabili necessarie per rappresentare un vettore dinamico di interi a 32 bit. In ottica OOP queste variabili rappresenteranno gli attributi della nostra classe. Utilizziamo una convenzione standard per i nomi delle variabili, ovvero il suffisso `_` per indicare che si tratta di variabili membro della nostra struttura.

### Funzioni per la struct

Il tipo di dato `vector_t`, rappresentativo della struttura `struct vector`, sarà l'entità base su cui verranno applicate tutte le funzioni di supporto. È una prassi molto comune nella programmazione C orientata agli oggetti, creare funzioni che operano su un tipo di dato specifico, passando un puntatore a quell'entità come primo parametro.

Le funzioni che dovremo definire saranno trasformate in metodi della nostra classe in ottica OOP, di conseguenza dovremo definire tutti le classiche funzioni che saranno necessarie a una classe orientata agli oggetti, a partire dal costruttore, distruttore, e tutte le funzioni di supporto per la gestione del vettore.

```c
// ...

vector_t *vector_constructor(vector_t *this) {
    this->size_ = 0;
    this->capacity_ = 0;
    this->data_ = NULL;
    return this;
}

vector_t *vector_destructor(vector_t *this) {
    free(this->data_);
    return this;
}

vector_t *vector_new(void) {
    vector_t *this = malloc(sizeof(vector_t));
    return vector_constructor(this);
}

void vector_delete(vector_t *this) {
    free(vector_destructor(this));
}

void vector_push_back(vector_t *this, int32_t number) {
    if (this->size_ >= this->capacity_) {
        size_t temp_capacity = this->capacity_ == 0 ? 1 : this->capacity_ * 2;
        int32_t *temp_data = realloc(this->data_, temp_capacity * sizeof(int32_t));
        this->data_ = temp_data;
        this->capacity_ = temp_capacity;
    }
    this->data_[this->size_] = number;
    this->size_++;
}

void vector_sort(vector_t *this) {
    qsort(this->data_, this->size_, sizeof(int32_t), compare_int32);
}

size_t vector_size(const vector_t *this) {
    return this->size_;
}

int32_t vector_at(const vector_t *this, size_t index) {
    assert(index < this->size_);
    return this->data_[index];
}

// ...
```

Le funzioni di supporto, per nostra convenzione e prassi comune, saranno tutte prefissate con il nome della struttura a cui appartengono, in questo caso `vector_`. In questo modo sarà chiaro a quale entità appartiene la funzione e sarà più semplice da leggere e comprendere.

Un aspetto interessante riguarda il costruttore e il distruttore. Queste funzioni non allocano e deallocano mai la memoria della struttura stessa, ma solamente della memoria interna della struttura. Questo è un aspetto molto importante da tenere a mente, in quanto la gestione della memoria della struttura stessa sarà demandata a funzioni specifiche, ovvero `vector_new` e `vector_delete`, oppure al chiamante delle funzioni `vector_constructor` e `vector_destructor`, che dovranno gestire la memoria della struttura stessa. Infatti, il costruttore si occupa solo di inizializzare le variabili membro della struttura, mentre il distruttore si occupa solo di liberare la memoria allocata per i dati interni della struttura.

Per allocare e deallocare la memoria della struttura è possibile utilizzare la via standard del C, ovvero `malloc` e `free`, oppure utilizzare le funzioni `vector_new` e `vector_delete`, che si occupano di allocare e deallocare la memoria della struttura stessa. In questo caso, verrà allocata e deallocata automaticamente la memoria per l'intera struttura, e all'interno delle funzioni stesse verranno chiamati il costruttore e il distruttore per inizializzare e liberare la memoria interna della struttura.

Alcune funzioni di supporto, come `vector_at` e `vector_sort`, definiscono il parametro `this` come `const vector_t *`. Questo è un aspetto molto importante, in quanto indica che la funzione non modificherà lo stato interno della struttura, e quindi non potrà modificare le variabili membro della struttura stessa. In questo modo, sarà possibile utilizzare queste funzioni anche su oggetti costanti, senza rischiare di modificare lo stato interno della struttura. Questo, oltre a essere utile al compilatore, risulta utile anche per il programmatore, in quanto indica chiaramente che la funzione non modificherà lo stato interno della struttura, e quindi non potrà modificare le variabili membro della struttura stessa.

All'interno della funzione `vector_at` viene utilizzata la funzione `assert` per verificare che l'indice passato come parametro sia valido, ovvero che sia compreso tra 0 e `size_ - 1`. In questo modo, se l'indice passato come parametro non è valido, il programma terminerà con un errore, evitando di accedere a memoria non valida e quindi di causare un comportamento indefinito. Notare come `index` non potrà mai essere negativo, in quanto il tipo di dato `size_t` è un tipo di dato senza segno, e quindi non può rappresentare valori negativi. Risulta invece interessante il fatto che `assert` viene verificato solamente in fase di debug, e quindi non sarà presente in fase di rilascio del programma.

### Pulizia del main

Applicando la struttura `struct vector` e le relative funzioni di supporto, possiamo semplificare notevolmente la funzione `main`.

```c
// ...

int main(int argc, char **argv) {
    if (argc != 3) {
        printf("Usage: %s <input_file> <output_file>\n", argv[0]);
        return EXIT_FAILURE;
    }

    const char *input_filename = argv[1];
    const char *output_filename = argv[2];

    FILE *input_file = fopen(input_filename, "r");
    if (input_file == NULL) {
        printf("Error: Could not open file %s\n", input_filename);
        return EXIT_FAILURE;
    }

    vector_t *vec = vector_new();

    int32_t number;
    while (fscanf(input_file, "%d", &number) == 1) {
        vector_push_back(vec, number);
    }
    fclose(input_file);

    vector_sort(vec);

    FILE *output_file = fopen(output_filename, "w");
    if (output_file == NULL) {
        vector_delete(vec);
        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    for (size_t index = 0; index < vector_size(vec); index++) {
        fprintf(output_file, "%d\n", vector_at(vec, index));
    }
    fclose(output_file);

    vector_delete(vec);
    return EXIT_SUCCESS;
}
```

In questo caso è stata utilizzata direttamente la funzione `vector_new` per allocare la memoria della struttura `vector_t`, e la funzione `vector_delete` per deallocare la memoria della struttura `vector_t`. Ma si poteva benissimo utilizzare anche `malloc` e `free` per ottenere lo stesso risultato.

```c
// ...

int main(int argc, char **argv) {
    if (argc != 3) {
        printf("Usage: %s <input_file> <output_file>\n", argv[0]);
        return EXIT_FAILURE;
    }

    const char *input_filename = argv[1];
    const char *output_filename = argv[2];

    FILE *input_file = fopen(input_filename, "r");
    if (input_file == NULL) {
        printf("Error: Could not open file %s\n", input_filename);
        return EXIT_FAILURE;
    }

    vector_t *vec = malloc(sizeof(vector_t));
    vector_constructor(vec);

    int32_t number;
    while (fscanf(input_file, "%d", &number) == 1) {
        vector_push_back(vec, number);
    }
    fclose(input_file);

    vector_sort(vec);

    FILE *output_file = fopen(output_filename, "w");
    if (output_file == NULL) {
        vector_destructor(vec);
        free(vec);
        printf("Error: Could not open file %s\n", output_filename);
        return EXIT_FAILURE;
    }

    for (size_t index = 0; index < vector_size(vec); index++) {
        fprintf(output_file, "%d\n", vector_at(vec, index));
    }
    fclose(output_file);

    vector_destructor(vec);
    free(vec);
    return EXIT_SUCCESS;
}
```

Oppure utilizzare semplicemente la struttura `struct vector` come variabile locale interna al `main`, lasciando quindi il compito di distruggere la variabile al compilatore, in quanto sarà memorizzata nello stack e non più nella heap memory.

```c
// ...

int main(int argc, char **argv) {
    if (argc != 3) {
        printf("Usage: %s <input_file> <output_file>\n", argv[0]);

    const char *input_filename = argv[1];
    const char *output_filename = argv[2];

    FILE *input_file = fopen(input_filename, "r");
    if (input_file == NULL) {
        printf("Error: Could not open file %s\n", input_filename);
        return EXIT_FAILURE;

    vector_t vec;
    vector_constructor(&vec);

    while (fscanf(input_file, "%d", &number) == 1) {
```
